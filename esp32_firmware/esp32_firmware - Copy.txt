// 4×4 Matrix Keypad — manual scan (no library needed)
const byte KEYPAD_ROWS = 4;
const byte KEYPAD_COLS = 4;
const byte KP_ROW[KEYPAD_ROWS] = {13, 12, 14, 27};
const byte KP_COL[KEYPAD_COLS] = {26, 25, 33, 32};
const char KP_MAP[KEYPAD_ROWS][KEYPAD_COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
char lastKey = 0;

void initKeypad() {
  for (int r = 0; r < KEYPAD_ROWS; r++) {
    pinMode(KP_ROW[r], OUTPUT);
    digitalWrite(KP_ROW[r], HIGH);
  }
  for (int c = 0; c < KEYPAD_COLS; c++) {
    pinMode(KP_COL[c], INPUT_PULLUP);
  }
  espPrintln("[KEYPAD] 4x4 matrix initialized.");
  espPrint("[KEYPAD] Rows: ");
  for (int r = 0; r < KEYPAD_ROWS; r++) { espPrint(String(KP_ROW[r])); if (r<3) espPrint(","); }
  espPrint("  Cols: ");
  for (int c = 0; c < KEYPAD_COLS; c++) { espPrint(String(KP_COL[c])); if (c<3) espPrint(","); }
  espPrintln("");
  for (int r = 0; r < KEYPAD_ROWS; r++) {
    espPrintln("[KEYPAD] Row D" + String(KP_ROW[r]) + "=" + String(digitalRead(KP_ROW[r])));
  }
  for (int c = 0; c < KEYPAD_COLS; c++) {
    espPrintln("[KEYPAD] Col D" + String(KP_COL[c]) + "=" + String(digitalRead(KP_COL[c])));
  }
}

char scanKeypad() {
  // Noise check: if any column reads LOW while ALL rows are HIGH,
  // there's a short or noise — reject all readings
  for (int r = 0; r < KEYPAD_ROWS; r++) digitalWrite(KP_ROW[r], HIGH);
  delay(2);
  for (int c = 0; c < KEYPAD_COLS; c++) {
    if (digitalRead(KP_COL[c]) == LOW) return 0;
  }

  for (int r = 0; r < KEYPAD_ROWS; r++) {
    digitalWrite(KP_ROW[r], LOW);
    delay(2);
    int lowCol = -1;
    int lowCount = 0;
    for (int c = 0; c < KEYPAD_COLS; c++) {
      if (digitalRead(KP_COL[c]) == LOW) {
        lowCol = c;
        lowCount++;
      }
    }
    if (lowCount == 1) {
      delay(2);
      if (digitalRead(KP_COL[lowCol]) == LOW) {
        digitalWrite(KP_ROW[r], HIGH);
        delay(1);
        return KP_MAP[r][lowCol];
      }
    }
    digitalWrite(KP_ROW[r], HIGH);
  }
  return 0;
}

void pollKeypad() {
  char k = scanKeypad();
  if (k && k != lastKey) {
    lastKey = k;
    lastKeyTime = millis();
    pendingKeypadKey = String(k);
  }
  if (!k && millis() - lastKeyTime > 200) lastKey = 0;
}

/* ═══════════════════════════════════════════════════════
 *  COMPLETE HARDWARE PINOUT — USE D NUMBERING
 * ═══════════════════════════════════════════════════════
 *
 *  MFRC522 (RFID) — SPI Bus (shared with MicroSD)
 *     SDA(SS)  →  D15   (SPI CS)
 *     SCK      →  D18   (SPI Clock)
 *     MOSI     →  D23   (SPI Master Out)
 *     MISO     →  D19   (SPI Master In)
 *     RST      →  D22
 *     3.3V     →  3.3V
 *     GND      →  GND
 *
 *  HW-084 (Clock / RTC Module) — I2C
 *     VCC      →  3.3V
 *     GND      →  GND
 *     SDA      →  D21   (shared with LCD)
 *     SCL      →  D22
 *
 *  FM10A (Fingerprint Sensor) — UART2 | VCC via MOSFET on D5
 *     FM10A RX →  D17   (ESP32 TX2)
 *     FM10A TX →  D16   (ESP32 RX2)
 *     T-out    →  D35   (TouchOut interrupt — DO NOT CHANGE)
 *     VCC      →  3.3V
 *     GND      →  GND
 *
 *  MicroSD Card — SPI Bus (shared with RFID)
 *     CS       →  D4
 *     SCK      →  D18   (shared)
 *     MOSI     →  D23   (shared)
 *     MISO     →  D19   (shared)
 *
 *  I2C LCD Display
 *     SDA      →  D21
 *     SCL      →  D22
 *     VCC      →  5V
 *     GND      →  GND
 *
 *  Indicator LED / Buzzer
 *     Signal   →  D2     (blinks on scan events; short beep on buzzer models)
 *     GND      →  GND
 *
 *  (FM10A VCC via MOSFET on D5 — power-cycle to recover from hangs)
 *
 *  Built-in LED
 *     BLUE_LED →  D2
 *
 * ═══════════════════════════════════════════════════════
 */

// ─── SD Card Data / Error / Power Logging ───────────────────
#define SD_LOG_DIR "/logs"

bool ensureLogDir() {
  if (!SD.exists(SD_LOG_DIR)) { if (!SD.mkdir(SD_LOG_DIR)) return false; }
  return true;
}

bool logToSD(const char *filename, const String &line) {
  if (!sdCardReady) {
    static unsigned long lastSdRetry = 0;
    if (millis() - lastSdRetry > 15000) {
      lastSdRetry = millis();
      pinMode(SD_CS_PIN, OUTPUT);
      digitalWrite(SD_CS_PIN, HIGH);
      SPI.begin(18, 19, 23, -1);
      SPI.setFrequency(1000000);
      sdCardReady = SD.begin(SD_CS_PIN, SPI, 1000000);
      #ifdef USE_FINGERPRINT
      if (sdCardReady) sdFpReady = true;
      #endif
    }
    if (!sdCardReady) return false;
  }
  if (!ensureLogDir()) return false;
  
  #ifdef USE_RFID
  digitalWrite(SS_PIN, HIGH); // deselect RFID reader to prevent SPI bus contention
  #endif
  
  File f = SD.open(String(SD_LOG_DIR) + "/" + filename, FILE_APPEND);
  if (!f) return false;
  f.println(line);
  f.close();
  return true;
}

// ─── Push a log entry to the server (if WiFi is up) ──────────
void logToServer(const String &logType, const String &source, const String &message) {
  if (WiFi.status() != WL_CONNECTED || !isOnline) return;
  WiFiClient c;
  HTTPClient h;
  String url = "http://" + SERVER_IP + "/csc2052/api/system_logs.php";
  h.begin(c, url);
  h.setTimeout(2000);
  h.addHeader("Content-Type", "application/x-www-form-urlencoded");
  String postData = "action=insert&log_type=" + logType +
                    "&source=" + source +
                    "&message=" + urlEncode(message) +
                    "&hardware_key=" + hardwareKey;
  int code = h.POST(postData);
  if (code <= 0) {
    espPrintln("[LOG] Server log push failed: " + String(code));
  }
  h.end();
}

// ─── Web: read a file from SD ───────────────────────────────
void handleSDFile() {
  String path = webServer.arg("path");
  if (path.length() == 0 || path.indexOf("..") >= 0) {
    webServer.send(400, "text/plain", "Bad path");
    return;
  }
  if (!SD.exists(path)) {
    webServer.send(404, "text/plain", "Not found: " + path);
    return;
  }
  File f = SD.open(path, FILE_READ);
  if (!f) { webServer.send(500, "text/plain", "Open failed"); return; }
  String content;
  while (f.available()) content += (char)f.read();
  f.close();
  webServer.send(200, "text/plain", content);
}

void lcdWrite(int row, String text) {
  if (row < 0 || row > 3) return;
  while (text.length() < LCD_COLS) text += ' ';
  if (text.length() > LCD_COLS) text = text.substring(0, LCD_COLS);
  lcdLine[row] = text;

#ifdef USE_LCD
  if (lcdOK) {
    restoreI2cForLcd();
    lcdPtr->setCursor(0, row);
    lcdPtr->print(text);
    // Don't reclaim RST here — it will be done when RFID is actually needed
  }
#endif

  espPrintln("[LCD] | " + text + " |");
}

void lcdClear() {
  String blank = "";
  while (blank.length() < LCD_COLS) blank += ' ';
  for (int i = 0; i < 4; i++) {
    lcdLine[i] = blank;
#ifdef USE_LCD
    if (lcdOK) {
      restoreI2cForLcd();
      lcdPtr->setCursor(0, i);
      lcdPtr->print(lcdLine[i]);
      reclaimRstForRfid();
    }
#endif
  }
  espPrintln("[LCD] | ------ CLEARED ------ |");
}

// ============================================================
// SYSTEM TIME SYNCHRONIZER (POSIX)
// ============================================================
void syncSystemTime(String timeStr) {
  int y, m, d, hh, mm, ss;
  if (sscanf(timeStr.c_str(), "%d-%d-%d %d:%d:%d", &y, &m, &d, &hh, &mm, &ss) == 6) {
    struct tm t;
    t.tm_year = y - 1900;
    t.tm_mon = m - 1;
    t.tm_mday = d;
    t.tm_hour = hh;
    t.tm_min = mm;
    t.tm_sec = ss;
    t.tm_isdst = -1;
    
    time_t epoch = mktime(&t);
    if (epoch != -1) {
      struct timeval tv;
      tv.tv_sec = epoch;
      tv.tv_usec = 0;
      settimeofday(&tv, NULL);
      timeSynced = true;
      espPrintln("[RTC] System clock synced to Server Time: " + timeStr);
    }
  }
}

// ─── URL-encode a string (for POST data) ────────────────────
String urlEncode(const String &s) {
  String out = "";
  for (unsigned int i = 0; i < s.length(); i++) {
    char c = s.charAt(i);
    if (isalnum(c) || c == '-' || c == '_' || c == '.' || c == '~') {
      out += c;
    } else if (c == ' ') {
      out += "+";
    } else {
      char buf[4];
      snprintf(buf, sizeof(buf), "%%%02X", (unsigned char)c);
      out += buf;
    }
  }
  return out;
}

String getTimestamp() {
  time_t now;
  struct tm timeinfo;
  time(&now);
  localtime_r(&now, &timeinfo);
  
  if (timeinfo.tm_year < 120) { // Unsynchronized clock year < 2020
#ifdef USE_HW084
    if (rtcOK) {
      restoreI2cForLcd();
      DateTime rtcNow = rtc.now();
      reclaimRstForRfid();
      char buf[25];
      snprintf(buf, sizeof(buf), "%04d-%02d-%02d %02d:%02d:%02d",
               rtcNow.year(), rtcNow.month(), rtcNow.day(),
               rtcNow.hour(), rtcNow.minute(), rtcNow.second());
      return String(buf);
    }
#endif
    unsigned long sec = millis() / 1000;
    unsigned long hh = sec / 3600; sec %= 3600;
    unsigned long mm = sec / 60;   sec %= 60;
    char buf[25];
    snprintf(buf, sizeof(buf), "BOOT+%02lu:%02lu:%02lu", hh, mm, sec);
    return String(buf);
  }
  
  char buf[25];
  strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", &timeinfo);
  return String(buf);
}

// ============================================================
// SD CARD CONFIG.TXT PARSER
// ============================================================
void readSDConfig() {
  if (!sdCardReady) {
    espPrintln("[SD-CONFIG] SD card not initialized. Using NVRAM defaults.");
    return;
  }
  if (!SD.exists("/config.txt")) {
    espPrintln("[SD-CONFIG] No config.txt on SD card. Using NVRAM defaults.");
    return;
  }
  File f = SD.open("/config.txt", FILE_READ);
  if (!f) {
    espPrintln("[SD-CONFIG] Cannot open config.txt. Using NVRAM defaults.");
    return;
  }
  espPrintln("[SD-CONFIG] Reading config.txt from SD card...");
  while (f.available()) {
    String line = f.readStringUntil('\n');
    line.trim();
    if (line.length() == 0 || line.startsWith("#")) continue;
    int eq = line.indexOf('=');
    if (eq < 0) continue;
    String key = line.substring(0, eq); key.trim();
    String val = line.substring(eq + 1); val.trim();
    if (val.length() == 0) continue;
    
    if      (key == "WIFI_SSID")    { WIFI_SSID = val; espPrintln("[SD-CONFIG]  SSID = " + val); }
    else if (key == "WIFI_PASS")    { WIFI_PASS = val; espPrintln("[SD-CONFIG]  Password set."); }
    else if (key == "SERVER_IP")    { SERVER_IP = val; espPrintln("[SD-CONFIG]  Server IP = " + val); }
    else if (key == "SERVER_URL")   { SERVER_IP = val; SERVER_IP.replace("http://", ""); SERVER_IP.replace("/csc2052", ""); SERVER_IP.replace("/", ""); espPrintln("[SD-CONFIG]  Server URL → IP: " + SERVER_IP); }
    else if (key == "DEVICE_NAME")  { DEVICE_NAME = val; espPrintln("[SD-CONFIG]  Device Name = " + val); }
    else if (key == "SECRET_KEY")   { secretKey = val; espPrintln("[SD-CONFIG]  Secret Key set."); }
    else if (key == "HARDWARE_KEY") { hardwareKey = val; espPrintln("[SD-CONFIG]  Hardware Key set."); }
    else if (key == "STATIC_DNS")     { staticDNS = val; espPrintln("[SD-CONFIG]  DNS = " + val); }
  }
  f.close();
  // wasAlreadyInit not needed — sdCardReady guards are sufficient

  
  // Save SD config to NVRAM for future boots
  preferences.begin("wifi", false);
  preferences.putString("ssid", WIFI_SSID);
  preferences.putString("pass", WIFI_PASS);
  preferences.end();
  preferences.begin("server", false);
  preferences.putString("ip", SERVER_IP);
  preferences.end();
  preferences.begin("config", false);
  if (DEVICE_NAME.length() > 0) preferences.putString("device_name", DEVICE_NAME);
  preferences.end();

bool tryConnectWifi() {
  if (WiFi.status() == WL_CONNECTED) return true;

  espPrint("[WIFI] Attempting connection to SSID: ");
  espPrintln(WIFI_SSID);
  
  WiFi.disconnect(true);
  delay(200);
  WiFi.mode(WIFI_STA);
  
  // Apply static IP if configured
  if (staticIP.length() > 0) {
    IPAddress ip, gw, sn, dns;
    bool hasDNS = false;
    if (ip.fromString(staticIP)) {
      if (staticGateway.length() > 0) gw.fromString(staticGateway); else gw = IPAddress(ip[0], ip[1], ip[2], 1);
      if (staticSubnet.length() > 0) sn.fromString(staticSubnet); else sn = IPAddress(255, 255, 255, 0);
      if (staticDNS.length() > 0) { dns.fromString(staticDNS); hasDNS = true; }
      if (hasDNS) {
        WiFi.config(ip, gw, sn, dns);
        espPrintln("[WIFI] Static IP: " + staticIP + " GW: " + gw.toString() + " SN: " + sn.toString() + " DNS: " + dns.toString());
      } else {
        WiFi.config(ip, gw, sn);
        espPrintln("[WIFI] Static IP: " + staticIP + " GW: " + gw.toString() + " SN: " + sn.toString());
      }
    }
  } else {
    espPrintln("[WIFI] Using DHCP.");
  }
  
  WiFi.begin(WIFI_SSID.c_str(), WIFI_PASS.c_str());

  unsigned long start = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - start < 8000) {
    delay(450);
    espPrint(".");
  }
  espPrintln();

  if (WiFi.status() == WL_CONNECTED) {
    espPrintln("[WIFI] Connected! Assigned IP: " + WiFi.localIP().toString());
    isOnline = true;
    flashLED(3, 80);
    return true;
  }
  
  espPrintln("[WIFI] Failed to connect. Running in OFFLINE mode.");
  isOnline = false;
  flashLED(1, 1000);
  return false;
}

void syncEnrollmentQueue() {
  if (WiFi.status() != WL_CONNECTED || !isOnline) return;
  if (!LittleFS.exists(ENROLL_QUEUE_FILE)) return;

  File f = LittleFS.open(ENROLL_QUEUE_FILE, FILE_READ);
  if (!f) return;

  espPrintln("\n[SYNC] Syncing offline enrollment queue...");
  std::vector<String> lines;
  while (f.available()) {
    String line = f.readStringUntil('\n');
    line.trim();
    if (line.length() > 3) lines.push_back(line);
  }
  f.close();

  if (lines.empty()) { LittleFS.remove(ENROLL_QUEUE_FILE); return; }

  std::vector<String> failed;
  for (auto& line : lines) {
    int c1 = line.indexOf(',');
    int c2 = line.indexOf(',', c1 + 1);
    if (c1 < 0 || c2 < 0) continue;

    String action = line.substring(0, c1);
    String stuNo  = line.substring(c1 + 1, c2);
    int slot     = line.substring(c2 + 1).toInt();

    WiFiClient client;
    HTTPClient http;
    http.begin(client, studentApiURL);
    http.addHeader("Content-Type", "application/x-www-form-urlencoded");

    String postData = "action=" + action + "&student_no=" + stuNo +
                      "&finger_id=" + String(slot) +
                      "&hardware_override=1" +
                      "&hardware_key=" + (hardwareKey.length() > 0 ? hardwareKey : "ss_hw_api_key_2052");
    int code = http.POST(postData);
    if (code == 200) {
      espPrintln("[SYNC] Enrollment synced: " + stuNo + " -> Slot #" + String(slot));
    } else {
      espPrintln("[SYNC] Enrollment sync failed: " + stuNo + " (HTTP " + String(code) + ")");
      failed.push_back(line);
    }
    http.end();
    delay(100);
  }

  if (failed.empty()) {
    LittleFS.remove(ENROLL_QUEUE_FILE);
    espPrintln("[SYNC] All enrollments synced and queue cleared.");
  } else {
    File fw = LittleFS.open(ENROLL_QUEUE_FILE, FILE_WRITE);
    if (fw) { for (auto& l : failed) fw.println(l); fw.close(); }
    espPrintln("[SYNC] " + String(failed.size()) + " enrollments pending retry.");
  }
}

int offlineQueueCount() {
  if (!LittleFS.exists(OFFLINE_LOG_FILE)) return 0;
  File f = LittleFS.open(OFFLINE_LOG_FILE, FILE_READ);
  if (!f) return 0;
  int count = 0;
  while (f.available()) {
    String line = f.readStringUntil('\n');
    if (line.length() > 3) count++;
  }
  f.close();
  return count;
}

void syncOfflineQueue() {
  if (WiFi.status() != WL_CONNECTED || !isOnline) {
    espPrintln("[SYNC] System offline. Offline sync aborted.");
    return;
  }
  if (!LittleFS.exists(OFFLINE_LOG_FILE)) return;

  File f = LittleFS.open(OFFLINE_LOG_FILE, FILE_READ);
  if (!f) return;

  espPrintln("\n[SYNC] Initiating background synchronizer queue...");
  std::vector<String> lines;
  while (f.available()) {
    String line = f.readStringUntil('\n');
    line.trim();
    if (line.length() > 3) lines.push_back(line);
  }
  f.close();

  if (lines.empty()) {
    LittleFS.remove(OFFLINE_LOG_FILE);
    return;
  }

  std::vector<String> failedLines;
  int successCount = 0;

  for (auto& line : lines) {
    int c1 = line.indexOf(',');
    int c2 = line.indexOf(',', c1 + 1);
    if (c1 < 0 || c2 < 0) continue;

    String fId       = line.substring(0, c1);
    String modality  = line.substring(c1 + 1, c2);
    String ts        = line.substring(c2 + 1);

    WiFiClient client;
    HTTPClient http;
    http.begin(client, serverURL);
    http.addHeader("Content-Type", "application/x-www-form-urlencoded");
    
    String tsEncoded = ts;
    tsEncoded.replace(" ", "%20");
    
    String postData = "finger_id=" + fId +
                      "&device_name=" + DEVICE_NAME +
                      "&course_code=" + currentCourse +
                      "&timestamp=" + tsEncoded +
                      "&modality=" + modality +
                      "&offline_sync=1" +
                      "&hardware_key=" + (hardwareKey.length() > 0 ? hardwareKey : "ss_hw_api_key_2052");

    int code = http.POST(postData);
    if (code > 0) {
      String resp = http.getString();
      // Only count as success if server accepted the record
      if (resp.indexOf("\"status\":\"ok\"") >= 0 || resp.indexOf("\"status\":\"updated\"") >= 0 || resp.indexOf("\"status\":\"duplicate\"") >= 0) {
        espPrintln("[SYNC] Sync success: Slot #" + fId + " -> " + resp);
        successCount++;
      } else {
        espPrintln("[SYNC] Sync rejected: Slot #" + fId + " -> " + resp);
        // Keep in queue for retry if server was unreachable, drop if valid rejection
        if (resp.indexOf("\"status\":\"not_enrolled\"") >= 0 || resp.indexOf("\"status\":\"no_session\"") >= 0) {
          espPrintln("[SYNC] Dropping record — " + resp);
        } else {
          failedLines.push_back(line);
        }
      }
    } else {
      espPrintln("[SYNC] Sync failed: Slot #" + fId + " -> Error " + http.errorToString(code));
      failedLines.push_back(line);
    }
    http.end();
    delay(100);
  }

  if (failedLines.empty()) {
    LittleFS.remove(OFFLINE_LOG_FILE);
    espPrintln("[SYNC] All " + String(successCount) + " queued records fully synced! Queue cleared.");
  } else {
    File fw = LittleFS.open(OFFLINE_LOG_FILE, FILE_WRITE);
    if (fw) {
      for (auto& fl : failedLines) fw.println(fl);
      fw.close();
    }
    espPrintln("[SYNC] Partial sync. Successfully Synced: " + String(successCount) + " | Pending: " + String(failedLines.size()));
  }
  // Also sync any queued offline enrollments
  syncEnrollmentQueue();
}

void sendHeartbeat() {
  if (WiFi.status() != WL_CONNECTED) {
    isOnline = false;
    return;
  }

  WiFiClient client;
  HTTPClient http;
  http.setTimeout(4000);
  String hbUrl = heartbeatURL + "?device=esp32&name=" + DEVICE_NAME + "&hardware_key=" + hardwareKey;
  http.begin(client, hbUrl);
  
  int code = http.GET();
  if (code > 0) {
    isOnline = true;
    String payload = http.getString();
    espPrint(".");
    
    // Server time syncing
    int timeIdx = payload.indexOf("\"time\":\"");
    if (timeIdx != -1) {
      String timeStr = payload.substring(timeIdx + 8, timeIdx + 27);
      syncSystemTime(timeStr);
    }
  } else {
    espPrint("\n[HTTP] Heartbeat Failed: " + http.errorToString(code));
    isOnline = false;
  }
  http.end();
}

// ============================================================
void markAttendance(String fingerId, String modality, String ts) {
  // Cooldown filter
  if (fingerId == lastScannedID && (millis() - lastScanTime < COOLDOWN_MS)) {
    espPrintln("[SYSTEM] Cooldown filtering duplicate scan for: " + fingerId);
    lcdWrite(2, "Already Logged!");
    delay(1000);
    printMenu();
    return;
  }
  lastScannedID = fingerId;
  lastScanTime = millis();

  if (WiFi.status() != WL_CONNECTED || !isOnline) {
    espPrintln("[OFFLINE] Server offline. Directing scan to LittleFS.");
    saveToOfflineQueue(fingerId, modality, ts);
    sessionScanned++;
    logAttendanceSD(fingerId, modality + "(offline)");
    indicatorBlink(3, 60);
    lcdWrite(2, "Logged Offline");
    delay(1200);
    printMenu();
    return;
  }

  WiFiClient client;
  HTTPClient http;
  http.setTimeout(6000);
  http.begin(client, serverURL);
  http.addHeader("Content-Type", "application/x-www-form-urlencoded");

  String tsEncoded = ts;
  tsEncoded.replace(" ", "%20");
  
  String postData = "finger_id=" + fingerId +
                    "&device_name=" + DEVICE_NAME +
                    "&course_code=" + currentCourse +
                    "&timestamp=" + tsEncoded +
                    "&modality=" + modality +
                    "&hardware_key=" + hardwareKey;

  int code = http.POST(postData);
  if (code > 0) {
    String payload = http.getString();
    espPrintln("[SERVER] Response Code " + String(code) + " | Payload: " + payload);
    isOnline = true;
    flashLED(2, 100);
    indicatorBlink(2, 80);

    // Dynamic Admin commands mapping
    if (payload.indexOf("\"status\":\"admin\"") >= 0 || payload.indexOf("\"status\": \"admin\"") >= 0) {
      espPrintln("\n[ADMIN] Administrator matched! Resetting dynamic lecture session.");
      lcdWrite(2, "Admin Matched!");
      currentCourse = "Unassigned Session";
      syncOfflineQueue();
      delay(1500);
      printMenu();
      http.end();
      return;
    }

    // Handle new session-based responses
    if (payload.indexOf("\"status\":\"not_enrolled\"") >= 0 || payload.indexOf("\"status\": \"not_enrolled\"") >= 0) {
      espPrintln("[ATTENDANCE] Student not enrolled in this course.");
      lcdWrite(2, "Not Enrolled!");
      delay(1500);
      printMenu();
      http.end();
      return;
    }
    if (payload.indexOf("\"status\":\"no_session\"") >= 0 || payload.indexOf("\"status\": \"no_session\"") >= 0) {
      espPrintln("[ATTENDANCE] No active lecture session.");
      lcdWrite(2, "No Active Lecture");
      delay(1500);
      printMenu();
      http.end();
      return;
    }
    if (payload.indexOf("\"status\":\"duplicate\"") >= 0 || payload.indexOf("\"status\": \"duplicate\"") >= 0) {
      espPrintln("[ATTENDANCE] Already recorded for today.");
      lcdWrite(2, "Already Logged!");
      delay(1000);
      printMenu();
      http.end();
      return;
    }

    // Success or updated — count and log
    sessionScanned++;
    logAttendanceSD(fingerId, modality);

    if (payload.indexOf("\"status\":\"updated\"") >= 0 || payload.indexOf("\"status\": \"updated\"") >= 0) {
      espPrintln("[ATTENDANCE] Re-scan — record updated.");
      lcdWrite(2, "Updated (Re-scan)");
    } else {
      lcdWrite(2, "Attendance Marked");
    }
  } else {
    espPrintln("[SERVER] POST Failure (" + http.errorToString(code) + "). Saved in queue.");
    saveToOfflineQueue(fingerId, modality, ts);
    isOnline = false;
    flashLED(1, 600);
  }
  http.end();
}

bool saveFingerprintToSD(uint16_t slot) {
  if (!sdFpReady) return false;
  if (finger.loadModel(slot) != FINGERPRINT_OK) return false;
  delay(100);
  if (finger.getModel() != FINGERPRINT_OK) return false;
  uint8_t buf[512];
  uint16_t offset = 0;
  while (true) {
    Adafruit_Fingerprint_Packet pkt(0, 0, NULL);
    if (finger.getStructuredPacket(&pkt, 2000) != FINGERPRINT_OK) break;
    uint16_t plen = pkt.length;
    if (plen > 2) plen -= 2;
    if (plen > 0) {
      if (offset + plen > 512) plen = 512 - offset;
      memcpy(buf + offset, pkt.data, plen);
      offset += plen;
    }
    if (pkt.type == 0x08) break;
  }
  if (offset < 16) return false;
  String path = String(FP_SD_DIR) + "/slot_" + String(slot) + ".dat";
  File f = SD.open(path, FILE_WRITE);
  if (!f) return false;
  f.write(buf, offset);
  f.close();
  espPrintln("[SD] Saved template slot " + String(slot) + " (" + String(offset) + " bytes)");
  return true;
}

bool uploadFingerprintToFM10A(uint16_t slot, const uint8_t *data, uint16_t len) {
  uint8_t downCmd[] = {0x0D, 0x01};
  Adafruit_Fingerprint_Packet cmdPkt(0x01, sizeof(downCmd), downCmd);
  finger.writeStructuredPacket(cmdPkt);
  Adafruit_Fingerprint_Packet ackPkt(0, 0, NULL);
  if (finger.getStructuredPacket(&ackPkt, 2000) != FINGERPRINT_OK) return false;
  if (ackPkt.length < 1 || ackPkt.data[0] != 0x00) return false;
  const int CHUNK = 32;
  uint16_t sent = 0;
  while (sent < len) {
    uint16_t chunk = (len - sent > CHUNK) ? CHUNK : (len - sent);
    bool isLast = (sent + chunk >= len);
    Adafruit_Fingerprint_Packet dpkt(isLast ? 0x08 : 0x02, chunk, (uint8_t*)(data + sent));
    finger.writeStructuredPacket(dpkt);
    if (finger.getStructuredPacket(&ackPkt, 3000) != FINGERPRINT_OK) return false;
    if (ackPkt.length < 1 || ackPkt.data[0] != 0x00) return false;
    sent += chunk;
  }
  return (finger.storeModel(slot) == FINGERPRINT_OK);
}

int importFingerprintFromSDIntoSlot(const char *filename, uint16_t dstSlot) {
  String path = String(FP_SD_DIR) + "/" + String(filename);
  File f = SD.open(path, FILE_READ);
  if (!f) return -1;
  uint8_t buf[512];
  uint16_t len = f.read(buf, sizeof(buf));
  f.close();
  if (len < 16) return -1;
  if (!uploadFingerprintToFM10A(dstSlot, buf, len)) return -1;
  return (int)dstSlot;
}

int findFreeFM10ASlot() {
  finger.getTemplateCount();
  if (finger.templateCount >= finger.capacity) return -1;
  for (int s = 1; s <= finger.capacity; s++) {
    if (finger.loadModel(s) != FINGERPRINT_OK) return s;
  }
  return -1;
}

int scanSDAndImport() {
  if (!sdFpReady) return -1;
  File dir = SD.open(FP_SD_DIR);
  if (!dir) return -1;
  int imported = 0;
  while (true) {
    File entry = dir.openNextFile();
    if (!entry) break;
    String name = entry.name();
    entry.close();
    if (!name.startsWith("slot_") || !name.endsWith(".dat")) continue;
    int s = name.substring(5, name.length() - 4).toInt();
    if (s <= 0) continue;
    if (finger.loadModel(s) == FINGERPRINT_OK) continue;
    int freeSlot = findFreeFM10ASlot();
    if (freeSlot < 0) { espPrintln("[FP] FM10A full, cannot import " + name); break; }
    if (importFingerprintFromSDIntoSlot(name.c_str(), freeSlot) >= 0) {
      espPrintln("[FP] Imported SD " + name + " to slot " + String(freeSlot));
      imported++;
    }
  }
  dir.close();
  return imported;
}

void runEnrollment(int overrideSlot) {
  if (!fpOK) {
    espPrintln("[ENROLL] ⚠ Fingerprint sensor not detected. Fix power (external 3.3V) and wiring.");
    lcdClear();
    lcdWrite(0, "FP SENSOR DOWN");
    lcdWrite(1, "Check power/wire");
    delay(2000);
    printMenu();
    return;
  }
  int slotId = overrideSlot;
  String studentNo = "";

  if (slotId == -1) {
    espPrintln("\n=============================================");
    espPrintln("        FINGERPRINT ENROLLMENT WIZARD        ");
    espPrintln("=============================================");
    espPrintln("Enter Student No via keypad (digits + # for /)");
    lcdClear();
    lcdWrite(0, "Enroll Student #");
    lcdWrite(1, "S/_");
    lcdWrite(2, "0-9=type #=/ D=OK");
    lcdWrite(3, "*=del A=cancel");
    while (Serial.available()) Serial.read();

    String enrollBuf = "";
    unsigned long waitStart = millis();
    while (studentNo == "") {
      webServer.handleClient();
      pollKeypad();
      if (pendingKeypadKey.length() > 0) {
        String key = pendingKeypadKey;
        pendingKeypadKey = "";
        if (key == "A") {
          espPrintln("[ENROLL] Cancelled via Keypad.");
          lcdWrite(1, "Cancelled");
          delay(800);
          return;
        }
        if (key == "*") {
          if (enrollBuf.length() > 0) {
            enrollBuf = enrollBuf.substring(0, enrollBuf.length() - 1);
            lcdWrite(1, "S/" + enrollBuf + "_");
          }
        } else if (key == "#") {
          enrollBuf += "/";
          lcdWrite(1, "S/" + enrollBuf + "_");
        } else if (key == "D") {
          if (enrollBuf.length() > 0) {
            studentNo = "S/" + enrollBuf;
          }
        } else if (key.length() == 1 && key.charAt(0) >= '0' && key.charAt(0) <= '9') {
          enrollBuf += key;
          lcdWrite(1, "S/" + enrollBuf + "_");
        }
      }
      if (Serial.available() > 0) {
        studentNo = Serial.readStringUntil('\n');
        studentNo.trim();
        if (studentNo.equalsIgnoreCase("cancel")) {
          espPrintln("[ENROLL] Cancelled by user.");
          return;
        }
        if (studentNo.length() < 2) studentNo = "";
      }
      if (millis() - waitStart > 120000) {
        espPrintln("[ENROLL] Input timeout. Aborting.");
        lcdWrite(1, "Timeout!");
        delay(1000);
        return;
      }
      delay(50);
    }
    espPrintln("[CONFIG] Enrolling Student: " + studentNo);

    if (isOnline && WiFi.status() == WL_CONNECTED) {
      espPrintln("[DB] Fetching next free database slot index...");
      WiFiClient client;
      HTTPClient http;
      http.setTimeout(6000);
      
      String queryUrl = studentApiURL + "?action=get_next_slot&hardware_override=1&hardware_key=" + hardwareKey;
      http.begin(client, queryUrl);
      
      int code = http.GET();
      if (code == 200) {
        String resp = http.getString();
        resp.trim();
        
        // Parse json response {"next_slot":8}
        int slotPos = resp.indexOf("\"next_slot\":");
        if (slotPos != -1) {
          int commaPos = resp.indexOf(",", slotPos);
          if (commaPos == -1) commaPos = resp.indexOf("}", slotPos);
          String slotStr = resp.substring(slotPos + 12, commaPos);
          slotStr.trim();
          slotId = slotStr.toInt();
        } else {
          slotId = resp.toInt();
        }
        
        if (slotId < 1 || slotId > 127) slotId = 1;
        espPrintln("[SERVER] Assigned biometric slot index: #" + String(slotId));
      } else {
        espPrintln("[SERVER] Database fetch failed (HTTP " + String(code) + "). Using Slot #1.");
        slotId = 1;
      }
      http.end();
    } else {
      espPrintln("[OFFLINE] Offline mode default. Storing in Slot #1.");
      slotId = 1;
    }
  } else {
    espPrintln("\n=============================================");
    espPrintln("      OTA TRIGGERED ENROLLMENT REQUEST       ");
    espPrintln("=============================================");
    espPrintln("[CONFIG] Assigned Slot Target: #" + String(slotId));
  }

  // ── Full-enrollment restart loop (max 3 attempts) ────────────
  int enrollAttempts = 0;
enroll_restart:
  enrollAttempts++;
  if (enrollAttempts > 3) {
    espPrintln("[ENROLL] Failed after 3 full attempts. Aborting.");
    lcdWrite(1, "Failed x3!");
    delay(1500);
    return;
  }
  if (enrollAttempts > 1) {
    espPrintln("\n[ENROLL] Full restart (attempt " + String(enrollAttempts) + "/3). Clean sensor and try again.");
    lcdClear();
    lcdWrite(0, ">> ENROLL #" + String(slotId));
    lcdWrite(1, "Attempt " + String(enrollAttempts) + "/3");
    delay(2000);
  }

  lcdClear();
  lcdWrite(0, ">> ENROLL SLOT #" + String(slotId));
  // ── First capture: wait for finger (180s, auto-retry) ──────
  unsigned long timeout = millis() + 180000;
  lcdWrite(1, "PLACE FINGER NOW");
  lcdWrite(3, "A/* cancel");
  while (finger.getImage() != FINGERPRINT_OK) {
    webServer.handleClient();
    pollKeypad();
    if (pendingKeypadKey.length() > 0) {
      String key = pendingKeypadKey;
      pendingKeypadKey = "";
      if (key == "A" || key == "*") {
        espPrintln("[ENROLL] Cancelled via Virtual Keypad (" + key + ").");
        lcdWrite(1, "Enroll Aborted!");
        delay(1200);
        return;
      }
    }
    if (Serial.available() > 0) {
      String in = Serial.readStringUntil('\n');
      in.trim();
      if (in.equalsIgnoreCase("cancel")) {
        espPrintln("[ENROLL] Cancelled.");
        lcdWrite(1, "Enroll Aborted!");
        delay(1200);
        return;
      }
    }
    if (millis() > timeout) {
      espPrintln("[ENROLL] Timeout waiting for sensor image (3 min). Touch sensor to retry.");
      lcdWrite(1, "Place finger...");
      lcdWrite(3, "A/* cancel");
      timeout = millis() + 180000;  // reset timeout — keep trying
      delay(50);
      continue;
    }
    delay(50);
  }

  espPrintln("[OK] First scan captured successfully!");
  lcdWrite(1, "Image captured!");
  flashLED(1, 100);

  // Convert first image to template (retry up to 3 times on poor quality)
  int tzRetries = 0;
  while (finger.image2Tz(1) != FINGERPRINT_OK) {
    tzRetries++;
    if (tzRetries >= 3) {
      espPrintln("[ERROR] Poor scan quality on first image after 3 attempts. Restarting.");
      lcdWrite(1, "Poor Quality x3!");
      delay(1500);
      goto enroll_restart;
    }
    espPrintln("[WARN] Poor scan quality (attempt " + String(tzRetries) + "/3). Clean sensor and retry.");
    lcdWrite(1, "Poor Quality!");
    lcdWrite(3, "Retry " + String(tzRetries) + "/3");
    delay(1000);
    lcdWrite(1, "Place finger again");
    // Wait for finger removal
    unsigned long rTimeout = millis() + 15000;
    while (finger.getImage() != FINGERPRINT_NOFINGER && millis() < rTimeout) { delay(50); }
    // Wait for re-placement
    rTimeout = millis() + 30000;
    while (finger.getImage() != FINGERPRINT_OK && millis() < rTimeout) {
      webServer.handleClient();
      delay(50);
    }
    if (millis() >= rTimeout) {
      espPrintln("[ENROLL] Timeout waiting for retry finger. Aborting.");
      lcdWrite(1, "Enroll Timeout!");
      delay(1200);
      return;
    }
  }

  // ── Additional scans (2..enrollCount) for better accuracy ──────
  // Buffer 1 has the first scan template
  for (int scan = 2; scan <= enrollCount; scan++) {
    // Wait for finger removal
    espPrintln("=> REMOVE FINGER (" + String(scan-1) + "/" + String(enrollCount) + " done)...");
    lcdWrite(1, "Remove finger");
    lcdWrite(3, "A/* cancel");
    delay(1000);
    unsigned long rTimeout = millis() + 60000;
    while (finger.getImage() != FINGERPRINT_NOFINGER) {
      webServer.handleClient();
      pollKeypad();
      if (pendingKeypadKey.length() > 0) {
        String key = pendingKeypadKey; pendingKeypadKey = "";
        if (key == "A" || key == "*") { espPrintln("[ENROLL] Cancelled."); lcdWrite(1, "Cancelled"); delay(800); return; }
      }
      if (Serial.available() > 0) { String in = Serial.readStringUntil('\n'); in.trim(); if (in.equalsIgnoreCase("cancel")) { espPrintln("[ENROLL] Cancelled."); return; } }
      if (millis() > rTimeout) { espPrintln("[ENROLL] Remove timeout. Proceeding..."); break; }
      delay(50);
    }

    // Wait for finger placement
    espPrintln("=> PLACE SAME FINGER (scan " + String(scan) + "/" + String(enrollCount) + ")");
    lcdWrite(1, "Scan " + String(scan) + "/" + String(enrollCount));
    timeout = millis() + 180000;
    while (finger.getImage() != FINGERPRINT_OK) {
      webServer.handleClient();
      pollKeypad();
      if (pendingKeypadKey.length() > 0) { String key = pendingKeypadKey; pendingKeypadKey = ""; if (key == "A" || key == "*") { espPrintln("[ENROLL] Cancelled."); return; } }
      if (Serial.available() > 0) { String in = Serial.readStringUntil('\n'); in.trim(); if (in.equalsIgnoreCase("cancel")) { espPrintln("[ENROLL] Cancelled."); return; } }
      if (millis() > timeout) { espPrintln("[ENROLL] Timeout. Try again."); timeout = millis() + 180000; delay(50); continue; }
      delay(50);
    }

    espPrintln("[OK] Scan " + String(scan) + " captured!");
    lcdWrite(1, "Captured!");
    flashLED(1, 100);

    // Convert to buffer 2 and merge
    int tzRetry = 0;
    while (finger.image2Tz(2) != FINGERPRINT_OK) {
      tzRetry++;
      if (tzRetry >= 3) {
        espPrintln("[ERROR] Poor quality x3. Restarting enrollment.");
        lcdWrite(1, "Poor Quality x3!");
        delay(1500);
        goto enroll_restart;
      }
      espPrintln("[WARN] Poor quality (attempt " + String(tzRetry) + "/3). Retrying scan " + String(scan) + "...");
      lcdWrite(3, "Retry " + String(tzRetry) + "/3");
      delay(1000);
      unsigned long rt = millis() + 15000;
      while (finger.getImage() != FINGERPRINT_NOFINGER && millis() < rt) { delay(50); }
      rt = millis() + 30000;
      while (finger.getImage() != FINGERPRINT_OK && millis() < rt) { webServer.handleClient(); delay(50); }
      if (millis() >= rt) { espPrintln("[ENROLL] Timeout retrying. Aborting."); lcdWrite(1, "Timeout!"); delay(1000); return; }
    }

    // Merge with existing model in buffer 1
    if (finger.createModel() != FINGERPRINT_OK) {
      espPrintln("FAILED. Fingerprints did not match (scan " + String(scan) + "/" + String(enrollCount) + "). Restarting.");
      lcdWrite(1, "No Match!");
      delay(1500);
      goto enroll_restart;
    }
    espPrintln("[OK] Scan " + String(scan) + "/" + String(enrollCount) + " merged!");
  }

  espPrintln("[OK] All " + String(enrollCount) + " scans enrolled successfully!");
  lcdWrite(1, "All " + String(enrollCount) + " done!");

  // Store model
  espPrint("Storing model in flash memory slot #" + String(slotId) + "... ");
  if (finger.storeModel(slotId) != FINGERPRINT_OK) {
    espPrintln("FAILED. Sensor memory write error.");
    lcdWrite(1, "Store Error!");
    delay(1500);
    return;
  }
  espPrintln("SUCCESS!");
  lcdWrite(1, "Enroll Success!");
  lcdWrite(2, "Slot #" + String(slotId));
  flashLED(3, 80);

  // Save template to SD card for backup/restore
  if (sdFpReady) saveFingerprintToSD(slotId);

  delay(2000);

  // Link slot ID to student number
  if (studentNo != "") {
    if (WiFi.status() == WL_CONNECTED && isOnline) {
      espPrintln("[DB] Linking Student '" + studentNo + "' to Slot #" + String(slotId) + " on Server...");
      WiFiClient client;
      HTTPClient http;
      http.setTimeout(6000);
      http.begin(client, studentApiURL);
      http.addHeader("Content-Type", "application/x-www-form-urlencoded");
      
      String postData = "action=link_student&student_no=" + studentNo +
                        "&finger_id=" + String(slotId) +
                        "&hardware_override=1" +
                        "&hardware_key=" + (hardwareKey.length() > 0 ? hardwareKey : "ss_hw_api_key_2052");

      int code = http.POST(postData);
      if (code == 200) {
        String resp = http.getString();
        resp.trim();
        espPrintln("[DB] Success! Link response: " + resp);
      } else {
        espPrintln("[DB] Server error linking (HTTP " + String(code) + "). Remember to link on dashboard.");
      }
      http.end();
    } else {
      saveEnrollmentToQueue(studentNo, slotId);
      lcdWrite(2, "Offline — queued");
      delay(1500);
    }
  } else {
    espPrintln("[INFO] Enrollment successful. Link Slot #" + String(slotId) + " manually via the Dashboard.");
  }
}

// ─── D22 reclaim/restore: switches between I2C SCL (LCD) and MFRC522 RST ───
#ifdef USE_RFID
void reclaimRstForRfid() {
  if (!i2cBusDirty) return; // Skip hardware reset if D22 SCL wasn't touched by I2C since last check
  
  // Force D22 as GPIO output LOW immediately (overrides any I2C peripheral)
  pinMode(RST_PIN, OUTPUT);
  digitalWrite(RST_PIN, LOW);
  Wire.end();
  delay(50);
  // Proper hardware reset sequence (matching working Test_frid)
  digitalWrite(RST_PIN, LOW);
  delay(100);
  digitalWrite(RST_PIN, HIGH);
  delay(100);
  
  // Deselect SD Card before initializing RFID to prevent contentions
  digitalWrite(SD_CS_PIN, HIGH);
  
  SPI.end();
  SPI.begin(18, 19, 23, -1);
  SPI.setFrequency(1000000); // 1 MHz stable SPI clock
  delay(10);
  mfrc522.PCD_Init();
  mfrc522.PCD_AntennaOn();
  // Verify the chip is alive
  byte ver = mfrc522.PCD_ReadRegister(MFRC522::VersionReg);
  if (ver == 0x00 || ver == 0xFF) {
    espPrintln("[RFID] ⚠ reclaimRstForRfid: MFRC522 NOT responding after reset!");
  }
  i2cBusDirty = false;
}

void restoreI2cForLcd() {
  i2cBusDirty = true;
  pinMode(RST_PIN, INPUT);
  delay(2);
  Wire.end();
  delay(2);
  Wire.begin(21, 22);
  delay(10);
  #ifdef USE_LCD
  if (lcdOK && lcdPtr) {
    lcdPtr->init();
    lcdPtr->backlight();
    for (int r = 0; r < 4; r++) {
      if (lcdLine[r].length() > 0) {
        lcdPtr->setCursor(0, r);
        lcdPtr->print(lcdLine[r]);
      }
    }
  }
  #endif
}

// ============================================================
// ATTENDANCE BIOMETRIC SCANNING ENGINE
// ============================================================
void scanForAttendance() {
  String ts = getTimestamp();

  // BUG FIX: Previously silently skipped RFID scan when no course was set.
  // Now shows a clear LCD message so the operator knows to set a course first.
  bool noCourse = (currentCourse.length() == 0 || currentCourse == "Unassigned Session" || currentCourse == "none");

#ifdef USE_RFID
  // Reclaim D22 from I2C before RFID read
  reclaimRstForRfid();

  // Still poll RFID when no course is set so the card read is not lost — just warn
  if (noCourse && enableRfid && mfrc522.PICC_IsNewCardPresent() && mfrc522.PICC_ReadCardSerial()) {
    mfrc522.PICC_HaltA();
    mfrc522.PCD_StopCrypto1();
    restoreI2cForLcd();
    espPrintln("[RFID] Card tapped but no active course session. Set course via START_COURSE or keypad A.");
    lcdWrite(2, "Set Course First!");
    lcdWrite(3, "A=pick course");
    { unsigned long _t = millis(); while (millis() - _t < 2000) { webServer.handleClient(); delay(20); } }
    printMenu();
    return;
  }

  if (noCourse) {
    restoreI2cForLcd();
    delay(50);
    return;
  }

  // RST is on D22 — reclaim from I2C before each read
  if (enableRfid && mfrc522.PICC_IsNewCardPresent() && mfrc522.PICC_ReadCardSerial()) {
    String rfidUid = "";
    for (byte i = 0; i < mfrc522.uid.size; i++) {
      if (mfrc522.uid.uidByte[i] < 0x10) rfidUid += "0";
      rfidUid += String(mfrc522.uid.uidByte[i], HEX);
    }
    rfidUid.toUpperCase();
    restoreI2cForLcd();
    espPrintln("\n" + ts + " [RFID] Card Read: " + rfidUid);
    lcdWrite(1, "Card: " + rfidUid);

    // Push UID to rfid_scanned.php for web dashboard polling
    if (WiFi.status() == WL_CONNECTED) {
      WiFiClient rClient;
      HTTPClient rHttp;
      rHttp.setTimeout(2000);
      rHttp.begin(rClient, "http://" + SERVER_IP + "/csc2052/api/rfid_scanned.php?uid=" + rfidUid + "&hardware_key=" + hardwareKey);
      rHttp.GET();
      rHttp.end();
    }

    // Feature 1: Smart RFID lookup — resolve UID to student name/number
    if (isOnline && WiFi.status() == WL_CONNECTED) {
      WiFiClient rfidClient;
      HTTPClient rfidHttp;
      rfidHttp.setTimeout(5000);
      rfidHttp.begin(rfidClient, coursesApiURL + "?action=lookup_rfid&uid=" + rfidUid + "&hardware_key=" + hardwareKey);
      int rfidCode = rfidHttp.GET();
      String rfidResp = "";
      if (rfidCode == 200) rfidResp = rfidHttp.getString();
      rfidHttp.end();
      rfidResp.trim();

      if (rfidResp.startsWith("ERR") || rfidResp.length() == 0) {
        espPrintln("[RFID] Unknown card UID: " + rfidUid);
        lcdWrite(2, "Unknown Card!");
        mfrc522.PICC_HaltA();
        mfrc522.PCD_StopCrypto1();
        // BUG FIX: was delay(2000) — blocked webServer.handleClient() so web commands were dropped
        { unsigned long _t = millis(); while (millis() - _t < 2000) { webServer.handleClient(); delay(20); } }
        printMenu();
        return;
      } else {
        // Parse "student_no|student_name"
        int pipeIdx = rfidResp.indexOf('|');
        String studentNo   = (pipeIdx > 0) ? rfidResp.substring(0, pipeIdx) : rfidResp;
        String studentName = (pipeIdx > 0) ? rfidResp.substring(pipeIdx + 1) : "";
        indicatorBlink(1, 60);
        espPrintln("[RFID] Resolved: " + studentNo + " (" + studentName + ")");
        lcdWrite(2, studentName.substring(0, LCD_COLS));
        lcdWrite(3, "Processing...");
        // markAttendance handles LCD line 2 with result
        markAttendance(studentNo, "rfid", ts);
        lcdWrite(3, "");
      }
    } else {
      // Offline fallback: use raw UID directly
      markAttendance(rfidUid, "rfid", ts);
    }

    mfrc522.PICC_HaltA();
    mfrc522.PCD_StopCrypto1();
    // BUG FIX: was delay(2000) — blocked webServer.handleClient() so web commands were dropped
    { unsigned long _t = millis(); while (millis() - _t < 2000) { webServer.handleClient(); delay(20); } }
    printMenu();
    return;
  }
#endif

#ifdef USE_FINGERPRINT
  if (!enableFingerprint || !fpOK) return;

  // Rate limit getImage() — 200ms
  static unsigned long lastFpPoll = 0;
  if (millis() - lastFpPoll < 200) return;
  lastFpPoll = millis();

  // Call getImage() every cycle — FM10A handles its own touch detection.
  // Returns FINGERPRINT_NOFINGER instantly (~20ms) when no finger present.
  int p = finger.getImage();

  // Track touch activity for power saving (VOUT OR actual image = touch)
  bool anyTouch = (p == FINGERPRINT_OK) || (digitalRead(FINGER_VOUT) == HIGH);
  if (anyTouch) {
    lastActivity = millis();
    if (lowPowerMode) {
      lowPowerMode = false;
      espPrintln("[POWER] Touch detected — normal mode");
    }
  } else if (!lowPowerMode && millis() - lastActivity > fpIdleThreshold) {
    lowPowerMode = true;
    espPrintln("[POWER] Idle — " + String(fpIdleThreshold/1000) + "s no touch");
    logPowerSD("LOW_POWER_ENTER");
  }

  if (p == FINGERPRINT_PACKETRECIEVEERR) {
    logErrorSD("FINGERPRINT", "Packet RX error");
    return;
  }

  if (p == FINGERPRINT_OK) {
    espPrintln("\n" + ts + " [FP] Touch detected. Capturing scan...");
    lcdWrite(2, "Analyzing scan...");
    flashLED(1, 50);

    if (finger.image2Tz() != FINGERPRINT_OK) {
      espPrintln("[FP] Scan quality too low. Try placing finger again.");
      lcdWrite(2, "Bad image quality");
      delay(1500);
      printMenu();
      return;
    }

    if (finger.fingerFastSearch() == FINGERPRINT_OK) {
      espPrintln("[FP] Match Found! Slot #" + String(finger.fingerID) + " Conf: " + String(finger.confidence));
      indicatorBlink(1, 60);
      lcdWrite(2, "Match! Slot #" + String(finger.fingerID));
      String fpStudentNo = String(finger.fingerID);
      if (WiFi.status() == WL_CONNECTED && isOnline) {
        WiFiClient fclient;
        HTTPClient fhttp;
        fhttp.setTimeout(4000);
        String fquery = coursesApiURL + "?action=lookup_fingerprint&slot=" + String(finger.fingerID) + "&hardware_key=" + hardwareKey;
        fhttp.begin(fclient, fquery);
        int fcode = fhttp.GET();
        if (fcode == 200) {
          String fresp = fhttp.getString();
          fresp.trim();
          int fpipe = fresp.indexOf('|');
          if (fpipe > 0) {
            fpStudentNo = fresp.substring(0, fpipe);
            String fpName = fresp.substring(fpipe + 1);
            lcdWrite(2, fpName.substring(0, LCD_COLS));
            lcdWrite(3, "Marked: " + fpStudentNo);
          }
        }
        fhttp.end();
      }
      markAttendance(fpStudentNo, "fingerprint", ts);
    } else {
      espPrintln("[FP] No match found. Trying SD card import...");
      lcdWrite(2, "No match. Check SD...");

      int imported = scanSDAndImport();
      if (imported > 0) {
        espPrintln("[FP] Imported " + String(imported) + " templates from SD. Retrying match...");
        lcdWrite(3, "Imported " + String(imported));
        if (finger.fingerFastSearch() == FINGERPRINT_OK) {
          espPrintln("[FP] Match Found (after SD import)! Slot #" + String(finger.fingerID));
          lcdWrite(2, "Match! Slot #" + String(finger.fingerID));
          String fpStudentNo2 = String(finger.fingerID);
          if (WiFi.status() == WL_CONNECTED && isOnline) {
            WiFiClient fclient;
            HTTPClient fhttp;
            fhttp.setTimeout(4000);
            String fquery = coursesApiURL + "?action=lookup_fingerprint&slot=" + String(finger.fingerID) + "&hardware_key=" + hardwareKey;
            fhttp.begin(fclient, fquery);
            int fcode = fhttp.GET();
            if (fcode == 200) {
              String fresp = fhttp.getString();
              fresp.trim();
              int fpipe = fresp.indexOf('|');
              if (fpipe > 0) {
                fpStudentNo2 = fresp.substring(0, fpipe);
                String fpName = fresp.substring(fpipe + 1);
                lcdWrite(2, fpName.substring(0, LCD_COLS));
                lcdWrite(3, "Marked: " + fpStudentNo2);
              }
            }
            fhttp.end();
          }
          markAttendance(fpStudentNo2, "fingerprint", ts);
        } else {
          espPrintln("[FP] Still no match after SD import. Access Denied.");
          lcdWrite(2, "Access Denied!");
          sessionDenied++;
          flashLED(1, 400);
        }
      } else {
        espPrintln("[FP] No new templates on SD. Access Denied.");
        lcdWrite(2, "Access Denied!");
        sessionDenied++;
        flashLED(1, 400);
      }
    }

    logPowerSD("LOW_POWER_EXIT");

    // Wait for finger release (check VOUT + getImage for robustness)
    espPrintln("=> Release finger from sensor glass...");
    delay(300);
    while (digitalRead(FINGER_VOUT) == HIGH) {
      webServer.handleClient();
      delay(50);
      if (finger.getImage() == FINGERPRINT_NOFINGER) break;
    }
    espPrintln(">> Waiting for next scan...");
    delay(1000);
    printMenu();
  }
#endif
}

// T9 MULTI-TAP ENGINE
// ============================================================
// Call this every loop() when a digit key 1-9/0 is pressed via keypad.
// After T9_TIMEOUT ms of no new tap on same key the char is committed.
void t9HandleKey(char k) {
  // Map key character to table index
  int idx = -1;
  if (k >= '1' && k <= '9') idx = k - '1';  // 0..8
  else if (k == '0')         idx = 9;
  if (idx < 0) return;

  unsigned long now = millis();
  if (k == t9LastKey && (now - t9LastTap) < T9_TIMEOUT) {
    // Same key, still within window — cycle to next char in set
    const char* charset = T9_MAP[idx];
    int len = strlen(charset);
    t9TapCount = (t9TapCount + 1) % len;
  } else {
    // New key OR timeout — commit previous char first
    if (t9Buffer.length() > 0 && t9LastKey != 0) {
      // already appended on previous press; nothing more to do
    }
    t9TapCount = 0;
  }

  t9LastKey = k;
  t9LastTap = now;

  // Build preview: replace last char if same key still cycling
  const char* charset = T9_MAP[idx];
  char chosen = charset[t9TapCount];

  // If this is a new key, append; otherwise replace last char
  if (t9LastKey == k && t9Buffer.length() > 0 && (millis() - t9LastTap) < T9_TIMEOUT) {
    // Check whether the last char in buffer belongs to this cycle
    // We track via t9TapCount: if tapCount==0 we appended fresh
    if (t9TapCount == 0) {
      t9Buffer += chosen; // fresh append
    } else {
      t9Buffer[t9Buffer.length()-1] = chosen; // replace last
    }
  } else {
    t9Buffer += chosen;
  }

  // Show live on LCD
  String preview = t9Buffer;
  if (preview.length() > 19) preview = preview.substring(preview.length()-19);
  lcdWrite(0, "Text input (T9):");
  lcdWrite(1, preview + "_");
  lcdWrite(2, String("Key:") + k + " Char:" + chosen);
  lcdWrite(3, "* del  D=confirm");
  espPrintln("[T9] Buffer: " + t9Buffer);
}

// Call every loop() to auto-commit T9 char when key is held
void t9Tick() {
  if (t9LastKey == 0) return;
  // Nothing extra needed — next keypress or D/confirm will read t9Buffer
}

void coursePickerKey(String key) {
  if (!courseMenuOpen) return;

  if (key == "B") {  // scroll up
    if (courseCount > 0) {
      courseMenuIdx = (courseMenuIdx - 1 + courseCount) % courseCount;
      lcdWrite(1, courseList[courseMenuIdx]);
      lcdWrite(0, String(courseMenuIdx+1) + "/" + String(courseCount) + " Course:");
      espPrintln("[PICKER] -> " + courseList[courseMenuIdx]);
    }
  } else if (key == "C") {  // scroll down
    if (courseCount > 0) {
      courseMenuIdx = (courseMenuIdx + 1) % courseCount;
      lcdWrite(1, courseList[courseMenuIdx]);
      lcdWrite(0, String(courseMenuIdx+1) + "/" + String(courseCount) + " Course:");
      espPrintln("[PICKER] -> " + courseList[courseMenuIdx]);
    }
  } else if (key == "D") {  // confirm
    String chosen;
    if (t9Buffer.length() > 0) {
      // User typed manually
      chosen = t9Buffer;
      t9Buffer = "";
    } else if (courseCount > 0) {
      chosen = courseList[courseMenuIdx];
    } else {
      espPrintln("[PICKER] Nothing to confirm.");
      return;
    }
    courseMenuOpen = false;
    currentCourse  = chosen;
    sessionScanned = 0;
    sessionDenied  = 0;
    sessionOffline = 0;
    espPrintln("[COURSE] Selected: " + currentCourse);
    espPrintln("[SESSION] Counters reset for new session.");
    espPrintln("[PICKER] Press A=Attendance  B=Enroll");
    lcdClear();
    lcdWrite(0, chosen.substring(0, LCD_COLS));
    lcdWrite(1, "A=Attendance");
    lcdWrite(2, "B=Enroll");
    lcdWrite(3, "*=cancel");
    awaitModeChoice = true;
  } else if (key == "*") {  // backspace / cancel
    if (t9Buffer.length() > 0) {
      t9Buffer = t9Buffer.substring(0, t9Buffer.length()-1);
      String preview = t9Buffer;
      if (preview.length() > 19) preview = preview.substring(preview.length()-19);
      lcdWrite(1, preview + "_");
      espPrintln("[T9] Backspace. Buffer: '" + t9Buffer + "'");
    } else {
      // cancel picker
      courseMenuOpen = false;
      t9Buffer = "";
      espPrintln("[PICKER] Cancelled.");
      lcdClear();
      lcdWrite(0, "Cancelled.");
      delay(800);
      printMenu();
    }
  } else if (key == "A") {  // switch to T9 typing mode
    courseCount = 0;  // hide list, force manual entry
    t9Buffer = "";
    t9LastKey = 0;
    lcdClear();
    lcdWrite(0, "Type Course Code");
    lcdWrite(1, "_");
    lcdWrite(2, "0-9=T9  *=del");
    lcdWrite(3, "D=OK  *=cancel");
    espPrintln("[PICKER] Switched to T9 manual entry.");
  } else {
    // Digit key — route to T9
    char k = key.charAt(0);
    if ((k >= '0' && k <= '9') || k == ' ') {
      t9HandleKey(k);
    }
  }
}

// ============================================================
// FINGERPRINT TEMPLATE BACKUP / RESTORE
// ============================================================
void backupAllFpTemplatesToServer() {
#ifdef USE_FINGERPRINT
  if (!fpOK) { espPrintln("[FP] Sensor not ready."); return; }
  if (!isOnline) { espPrintln("[FP] Offline — cannot backup."); return; }

  espPrintln("\n[FP-BACKUP] Starting template backup to server...");
  lcdClear();
  lcdWrite(0, "FP Backup...");

  finger.getTemplateCount();
  uint16_t total = finger.templateCount;
  espPrintln("[FP-BACKUP] Sensor reports " + String(total) + " templates.");

  int saved = 0, failed = 0;
  for (int slot = 1; slot <= 127; slot++) {
    if (finger.loadModel(slot) != FINGERPRINT_OK) continue;  // slot empty

    // Download raw template from sensor
    while (mySerial.available()) mySerial.read(); // Flush RX buffer
    if (finger.getModel() != FINGERPRINT_OK) { failed++; continue; }

    // Read the template packets from serial (2 packets * 267 bytes = 534 bytes)
    uint8_t bytesReceived[534];
    memset(bytesReceived, 0xff, 534);
    uint32_t starttime = millis();
    int readBytes = 0;
    while (readBytes < 534 && (millis() - starttime) < 2000) {
      if (mySerial.available()) {
        bytesReceived[readBytes++] = mySerial.read();
      } else {
        delay(1);
      }
    }

    if (readBytes < 534) {
      failed++;
      espPrintln("[FP-BACKUP] Slot " + String(slot) + " FAILED: Timeout reading serial template data (" + String(readBytes) + " bytes read)");
      continue;
    }

    // Filter only the data packets (256 bytes payload each)
    uint8_t fingerTemplate[512];
    memset(fingerTemplate, 0xff, 512);

    int uindx = 9, index = 0;
    memcpy(fingerTemplate + index, bytesReceived + uindx, 256);
    uindx += 256; // skip data
    uindx += 2;   // skip checksum
    uindx += 9;   // skip next header
    index += 256; // advance pointer
    memcpy(fingerTemplate + index, bytesReceived + uindx, 256);

    // Build hex string from fingerTemplate (512 bytes)
    String hex = "";
    for (int i = 0; i < 512; i++) {
      char buf[3];
      sprintf(buf, "%02X", fingerTemplate[i]);
      hex += buf;
    }

    // Lookup which student owns this slot
    WiFiClient client;
    HTTPClient http;
    http.setTimeout(6000);
    http.begin(client, coursesApiURL +
      "?action=get_fp_template&slot=" + String(slot) +
      "&hardware_key=" + hardwareKey);
    http.GET();
    String resp = http.getString();
    http.end();
    String studentNo = "";
    if (!resp.startsWith("ERR")) {
      // resp = "student_no|hexblob"
      int pipe = resp.indexOf('|');
      if (pipe > 0) studentNo = resp.substring(0, pipe);
    }
    if (studentNo == "") studentNo = "SLOT" + String(slot);

    // POST to server
    WiFiClient c2;
    HTTPClient http2;
    http2.setTimeout(8000);
    http2.begin(c2, coursesApiURL);
    http2.addHeader("Content-Type", "application/x-www-form-urlencoded");
    String body = "action=save_fp_template&hardware_key=" + hardwareKey;
    body += "&slot=" + String(slot);
    body += "&student_no=" + studentNo;
    body += "&hex=" + hex;
    int code = http2.POST(body);
    resp = http2.getString();
    http2.end();

    if (code == 200 && resp.startsWith("OK")) {
      saved++;
      espPrintln("[FP-BACKUP] Slot " + String(slot) + " -> " + studentNo + " OK");
    } else {
      failed++;
      espPrintln("[FP-BACKUP] Slot " + String(slot) + " FAILED: " + resp);
    }
    lcdWrite(1, "Slot " + String(slot) + "/127");
    lcdWrite(2, String(saved) + " saved " + String(failed) + " fail");
    webServer.handleClient();
    delay(50);
  }
  espPrintln("[FP-BACKUP] Done. " + String(saved) + " saved, " + String(failed) + " failed.");
  lcdWrite(0, "Backup Done!");
  lcdWrite(1, String(saved) + " templates saved");
  delay(2000);
  printMenu();
#else
  espPrintln("[FP-BACKUP] Fingerprint not compiled in.");
#endif
}

void restoreFpTemplatesFromServer() {
#ifdef USE_FINGERPRINT
  if (!fpOK) { espPrintln("[FP] Sensor not ready."); return; }
  if (!isOnline) { espPrintln("[FP] Offline — cannot restore."); return; }

  espPrintln("\n[FP-RESTORE] Fetching slot list from server...");
  lcdClear();
  lcdWrite(0, "FP Restore...");

  WiFiClient client;
  HTTPClient http;
  http.setTimeout(5000);
  http.begin(client, coursesApiURL +
    "?action=list_fp_slots&hardware_key=" + hardwareKey);
  int code = http.GET();
  String body = http.getString();
  http.end();

  if (code != 200 || body.startsWith("ERR")) {
    espPrintln("[FP-RESTORE] Failed to get slot list: " + body);
    return;
  }

  int restored = 0, failed = 0;
  // body = "1:S/20/001,2:S/20/002,..."
  int pos = 0;
  while (pos < (int)body.length()) {
    int comma = body.indexOf(',', pos);
    String entry = (comma == -1) ? body.substring(pos) : body.substring(pos, comma);
    entry.trim();
    pos = (comma == -1) ? body.length() : comma + 1;
    if (entry.length() == 0) continue;

    int colon = entry.indexOf(':');
    if (colon < 0) continue;
    int slot = entry.substring(0, colon).toInt();
    if (slot < 1 || slot > 127) continue;

    // Fetch the hex template
    WiFiClient c2;
    HTTPClient http2;
    http2.setTimeout(8000);
    http2.begin(c2, coursesApiURL +
      "?action=get_fp_template&slot=" + String(slot) +
      "&hardware_key=" + hardwareKey);
    int code2 = http2.GET();
    String resp2 = http2.getString();
    http2.end();

    if (code2 != 200 || resp2.startsWith("ERR")) {
      failed++;
      espPrintln("[FP-RESTORE] Slot " + String(slot) + " no data.");
      continue;
    }

    // resp2 = student_no|hexblob
    int pipe = resp2.indexOf('|');
    if (pipe < 0) { failed++; continue; }
    String hexBlob = resp2.substring(pipe + 1);
    if (hexBlob.length() < 1024) { failed++; continue; } // must be 512 bytes = 1024 hex chars

    // Convert hex -> bytes
    uint8_t fingerTemplate[512];
    memset(fingerTemplate, 0xff, 512);
    for (int i = 0; i < 512 && (i*2+1) < (int)hexBlob.length(); i++) {
      char hi = hexBlob.charAt(i*2);
      char lo = hexBlob.charAt(i*2+1);
      auto hexDigit = [](char c) -> uint8_t {
        if (c >= '0' && c <= '9') return c - '0';
        if (c >= 'A' && c <= 'F') return c - 'A' + 10;
        if (c >= 'a' && c <= 'f') return c - 'a' + 10;
        return 0;
      };
      fingerTemplate[i] = (hexDigit(hi) << 4) | hexDigit(lo);
    }

    // Download template to sensor's CharBuffer1 via DownChar command
    while (mySerial.available()) mySerial.read(); // Flush RX buffer
    
    uint8_t downCharData[] = {0x09, 0x01}; // 0x09 is DownChar instruction, 0x01 is buffer 1
    Adafruit_Fingerprint_Packet downCharPacket(FINGERPRINT_COMMANDPACKET, sizeof(downCharData), downCharData);
    finger.writeStructuredPacket(downCharPacket);
    
    if (finger.getStructuredPacket(&downCharPacket) != FINGERPRINT_OK || downCharPacket.data[0] != FINGERPRINT_OK) {
      failed++;
      espPrintln("[FP-RESTORE] Slot " + String(slot) + " DownChar failed.");
      continue;
    }

    // Transmit template packets (first packet FINGERPRINT_DATAPACKET, second packet FINGERPRINT_ENDDATAPACKET)
    auto writeTemplatePacket = [&](uint8_t packetType, const uint8_t *data256) {
      mySerial.write(0xEF);
      mySerial.write(0x01);
      mySerial.write(0xFF);
      mySerial.write(0xFF);
      mySerial.write(0xFF);
      mySerial.write(0xFF);
      mySerial.write(packetType);
      mySerial.write(0x01);
      mySerial.write(0x02);
      
      uint16_t sum = packetType + 0x01 + 0x02;
      for (int i = 0; i < 256; i++) {
        mySerial.write(data256[i]);
        sum += data256[i];
      }
      mySerial.write((uint8_t)(sum >> 8));
      mySerial.write((uint8_t)(sum & 0xFF));
    };

    writeTemplatePacket(FINGERPRINT_DATAPACKET, fingerTemplate);
    writeTemplatePacket(FINGERPRINT_ENDDATAPACKET, fingerTemplate + 256);

    // Read the acknowledge packet from DownChar raw data transfer
    if (finger.getStructuredPacket(&downCharPacket) != FINGERPRINT_OK || downCharPacket.data[0] != FINGERPRINT_OK) {
      failed++;
      espPrintln("[FP-RESTORE] Slot " + String(slot) + " raw data transfer rejected by sensor.");
      continue;
    }

    if (finger.storeModel(slot) == FINGERPRINT_OK) {
      restored++;
      espPrintln("[FP-RESTORE] Slot " + String(slot) + " restored.");
    } else {
      failed++;
      espPrintln("[FP-RESTORE] Slot " + String(slot) + " store FAILED.");
    }
    lcdWrite(1, "Slot " + String(slot));
    lcdWrite(2, String(restored) + " OK " + String(failed) + " fail");
    webServer.handleClient();
    delay(50);
  }
  espPrintln("[FP-RESTORE] Done. " + String(restored) + " restored, " + String(failed) + " failed.");
  lcdWrite(0, "Restore Done!");
  lcdWrite(1, String(restored) + " slots loaded");
  delay(2000);
  printMenu();
#else
  espPrintln("[FP-RESTORE] Fingerprint not compiled in.");
#endif
}

// ASK COURSE THEN START ATTENDANCE — called by 'A' key
// ============================================================
void askCourseAndStartAttendance() {
  espPrintln("\n=============================================");
  espPrintln("      START BACKGROUND ATTENDANCE SCAN       ");
  espPrintln("=============================================");
  if (currentCourse.length() > 0) {
    espPrintln("[INFO] Current active course: " + currentCourse);
    espPrintln(">> Press ENTER to keep it, or type a new Course Code:");
  } else {
    espPrintln(">> Enter Course Code to record attendance (e.g. CSC2052):");
    espPrintln("   (Type 'cancel' to abort)");
  }

  while (Serial.available()) Serial.read(); // flush buffer

  String input = "";
  unsigned long waitStart = millis();
  // Wait up to 30 seconds for input
  while (millis() - waitStart < 30000) {
    if (Serial.available() > 0) {
      input = Serial.readStringUntil('\n');
      input.trim();
      break;
    }
    delay(50);
  }

  // Timeout — just keep current course if any
  if (input.length() == 0) {
    if (currentCourse.length() > 0) {
      espPrintln("[INFO] No input — keeping course: " + currentCourse);
    } else {
      espPrintln("[WARN] No course entered and none active. Aborting.");
      return;
    }
  } else if (input.equalsIgnoreCase("cancel")) {
    espPrintln("[INFO] Cancelled. No mode change.");
    return;
  } else {
    currentCourse = input;
    espPrintln("[COURSE] Set to: " + currentCourse);
    flashLED(2, 80);
  }

  currentMode = "ATTENDANCE_MODE";
  espPrintln("[MODE] >> Background Attendance Scanning STARTED.");
  espPrintln("[MODE] >> Course: " + currentCourse);
  espPrintln("[MODE] >> Place finger or scan RFID to mark attendance.");
  lcdClear();
  lcdWrite(0, "ATTENDANCE SCAN");
  lcdWrite(1, currentCourse.substring(0, LCD_COLS));
  lcdWrite(2, isOnline ? "ONLINE" : "OFFLINE");
  lcdWrite(3, "Ready for scan...");
  printMenu();
}

// ============================================================
// HTTP WebServer CONTROLS & OTA INJECTORS
// ============================================================
void handleDashboardCommand() {
  webServer.sendHeader("Access-Control-Allow-Origin", "*");
  webServer.sendHeader("Access-Control-Allow-Methods", "GET, OPTIONS");
  webServer.sendHeader("Access-Control-Allow-Headers", "*");

  if (!webServer.hasArg("command")) {
    webServer.send(400, "application/json", "{\"error\":\"Missing command\"}");
    return;
  }

  String cmd = webServer.arg("command");
  espPrintln("\n[OTA Web Server] Received command request: " + cmd);

  // 1. PING
  if (cmd == "PING") {
    String resp = "{\"status\":\"PONG\",\"mode\":\"" + currentMode +
                  "\",\"course\":\"" + currentCourse +
                  "\",\"online\":" + (isOnline ? "true" : "false") +
                  ",\"queue\":" + String(offlineQueueCount()) + "}";
    webServer.send(200, "application/json", resp);
    return;
  }

  // 2. ATTENDANCE_MODE
  if (cmd == "ATTENDANCE_MODE") {
    currentMode = "ATTENDANCE_MODE";
    pendingEnrollSlot = -1;
    espPrintln("[OTA] Mode updated to ATTENDANCE_MODE");
    webServer.send(200, "text/plain", "ACK: Mode set to ATTENDANCE_MODE");
    printMenu();
    return;
  }

  // 3. ENROLL_MODE
  if (cmd == "ENROLL_MODE") {
    currentMode = "ENROLL_MODE";
    espPrintln("[OTA] Mode updated to ENROLL_MODE");
    webServer.send(200, "text/plain", "ACK: Mode set to ENROLL_MODE");
    printMenu();
    return;
  }

  // 4. START_COURSE  — also force attendance mode
  if (cmd.startsWith("START_COURSE ")) {
    currentCourse = cmd.substring(13);
    currentCourse.trim();
    currentMode = "ATTENDANCE_MODE";
    espPrintln("[OTA] Course set: " + currentCourse + " | Mode: ATTENDANCE");
    webServer.send(200, "text/plain", "ACK: Course=" + currentCourse + " Mode=ATTENDANCE");
    printMenu();
    return;
  }

  // 5. ENROLL <slot>
  if (cmd.startsWith("ENROLL ")) {
    String slotStr = cmd.substring(7);
    slotStr.trim();
    pendingEnrollSlot = slotStr.toInt();
    if (pendingEnrollSlot < 1 || pendingEnrollSlot > 127) {
      webServer.send(400, "text/plain", "ERR: Invalid biometric slot allocation (1-127)");
      pendingEnrollSlot = -1;
      return;
    }
    currentMode = "ENROLL_MODE";
    espPrintln("[OTA] Remote Web-Triggered Enrollment queued for Slot #" + String(pendingEnrollSlot));
    webServer.send(200, "text/plain", "ACK: Enrollment queued for Slot " + slotStr);
    return;
  }

  // 5.5. SCAN_RFID — enter RFID scan mode for web enrollment
  if (cmd == "SCAN_RFID") {
    pendingRfidScan = true;
    webServer.send(200, "text/plain", "ACK: SCAN_RFID");
    espPrintln("[OTA] SCAN_RFID received. Tap card within 30s.");
    return;
  }

  // 5.6. BEEP — flash indicator LED and fire buzzer (original behavior)
  if (cmd == "BEEP") {
    buzzerBeep(150);
    delay(80);
    buzzerBeep(150);
    webServer.send(200, "text/plain", "ACK: BEEP");
    espPrintln("[OTA] BEEP");
    return;
  }

  // 6. SETWIFI <ssid>|<pass>[|<identity>][|<method>|<token>]
  if (cmd.startsWith("SETWIFI ")) {
    String params = cmd.substring(8);
    int p1 = params.indexOf('|');
    String newSsid = (p1 >= 0) ? params.substring(0, p1) : params;
    String rest    = (p1 >= 0) ? params.substring(p1 + 1) : "";
    int p2 = rest.indexOf('|');
    String newPass = (p2 >= 0) ? rest.substring(0, p2) : rest;
    String rest2 = (p2 >= 0) ? rest.substring(p2 + 1) : "";
    int p3 = rest2.indexOf('|');
    String identity = (p3 >= 0) ? rest2.substring(0, p3) : "";
    String encMethod = "";
    String token = "";
    if (p3 >= 0) {
      String after = rest2.substring(p3 + 1);
      int p4 = after.indexOf('|');
      if (p4 >= 0) {
        encMethod = after.substring(0, p4);
        token = after.substring(p4 + 1);
      } else {
        encMethod = after;
      }
    }
    
    espPrintln("[OTA] WiFi update -> SSID: " + newSsid + (identity != "" ? " (Enterprise: " + identity + ")" : ""));
    
    preferences.begin("wifi", false);
    preferences.putString("ssid", newSsid);
    preferences.putString("pass", newPass);
    if (identity != "") preferences.putString("identity", identity);
    else preferences.putString("identity", "");
    preferences.end();
    
    if (encMethod != "") {
      preferences.begin("security", false);
      preferences.putString("method", encMethod);
      if (token != "") preferences.putString("token", token);
      preferences.end();
      espPrintln("[OTA] Security -> Method: " + encMethod);
    }
    
    webServer.send(200, "text/plain", "ACK: WiFi saved. Rebooting...");
    delay(1000);
    ESP.restart();
    return;
  }

  // 7. REBOOT
  if (cmd == "REBOOT") {
    webServer.send(200, "text/plain", "ACK: Restarting DevKit...");
    delay(500);
    ESP.restart();
    return;
  }

  // 8. CLEARLOGS
  if (cmd == "CLEARLOGS") {
    clearOfflineQueue();
    webServer.send(200, "text/plain", "ACK: Offline logs cleared.");
    return;
  }

  // 9. DUMP_OFFLINE
  if (cmd == "DUMP_OFFLINE") {
    if (!LittleFS.exists(OFFLINE_LOG_FILE)) {
      webServer.send(200, "text/plain", "ACK: Offline queue is empty.");
      return;
    }
    File f = LittleFS.open(OFFLINE_LOG_FILE, FILE_READ);
    String resp = "Offline Queue (" + String(offlineQueueCount()) + " records):\n";
    while (f.available()) {
      resp += f.readStringUntil('\n') + "\n";
    }
    f.close();
    webServer.send(200, "text/plain", resp);
    return;
  }

  // 10. SYNC_OFFLINE
  if (cmd == "SYNC_OFFLINE") {
    espPrintln("[OTA] Web requested manual background offline sync.");
    syncOfflineQueue();
    webServer.send(200, "text/plain", "ACK: Offline queue sync triggered.");
    return;
  }

  // 11. FORMAT_FS
  if (cmd == "FORMAT_FS") {
    espPrintln("[OTA] Web request: Formatting internal flash partition LittleFS...");
    LittleFS.format();
    webServer.send(200, "text/plain", "ACK: Flash formatted. Rebooting DevKit...");
    delay(500);
    ESP.restart();
    return;
  }

  // 12. SYNC_TIME
  if (cmd == "SYNC_TIME") {
    espPrintln("[OTA] Web sync clock requested. Force syncing...");
    if (WiFi.status() == WL_CONNECTED) {
      sendHeartbeat();
      webServer.send(200, "text/plain", "ACK: Clock synced to server time: " + getTimestamp());
    } else {
      webServer.send(200, "text/plain", "ERR: Network offline, sync failed.");
    }
    return;
  }

  // 13. WIPE_FP_TEMPLATES
  if (cmd == "WIPE_FP_TEMPLATES") {
#ifdef USE_FINGERPRINT
    espPrintln("[OTA] Web request: Wipe all fingerprint templates.");
    if (finger.emptyDatabase() == FINGERPRINT_OK) {
      espPrintln("[OTA] Sensor database wiped.");
      webServer.send(200, "text/plain", "ACK: Biometrics templates database cleared successfully.");
    } else {
      webServer.send(200, "text/plain", "ERR: Sensor flash write block error.");
    }
#else
    webServer.send(200, "text/plain", "ERR: Fingerprint compilation flag not defined.");
#endif
    return;
  }

  // 14. GETLCD
  if (cmd == "GETLCD") {
    String resp = lcdLine[0] + "|" + lcdLine[1] + "|" + lcdLine[2] + "|" + lcdLine[3];
    webServer.send(200, "text/plain", resp);
    return;
  }

  // 15. LCD <row> <text>
  if (cmd.startsWith("LCD ")) {
    String args = cmd.substring(4);
    int spacePos = args.indexOf(' ');
    if (spacePos > 0) {
      int row = args.substring(0, spacePos).toInt();
      String text = args.substring(spacePos + 1);
      if (row >= 0 && row <= 3) {
        lcdWrite(row, text);
        webServer.send(200, "text/plain", "ACK: LCD row " + String(row) + " updated.");
      } else {
        webServer.send(400, "text/plain", "ERR: Row index error.");
      }
    } else {
      webServer.send(400, "text/plain", "ERR: Format 'LCD <row 0-3> <text>'.");
    }
    return;
  }

  // 16. GETSTATUS
  if (cmd == "GETSTATUS") {
    int rssi = WiFi.RSSI();
    uint32_t uptimeSec = millis() / 1000;
    int queueSize = offlineQueueCount();
    
    String resp = "";
    resp += "STATUS: " + String(isOnline ? "ONLINE" : "OFFLINE") + "\n";
    resp += "MODE: " + currentMode + "\n";
    resp += "COURSE: " + currentCourse + "\n";
    resp += "SENSORS: " + String(enableFingerprint ? "FP " : "") + String(enableRfid ? "RFID " : "") + String(enableLcd ? "LCD " : "") + String(enableHw084 ? "HW084 " : "") + String(enableFace ? "FACE " : "") + String(requireMfa ? "| MFA" : "") + "\n";
    resp += "RSSI: " + String(rssi) + "\n";
    resp += "UPTIME: " + String(uptimeSec) + "\n";
    resp += "QUEUE: " + String(queueSize) + "\n";
    resp += "FS_USED: " + String(LittleFS.usedBytes()) + "\n";
    resp += "FS_TOTAL: " + String(LittleFS.totalBytes()) + "\n";
    resp += "HEAP: " + String(ESP.getFreeHeap()) + "\n";
    resp += "DEVICE: " + DEVICE_NAME + "\n";
    resp += "IP: " + WiFi.localIP().toString() + "\n";
    resp += "SESSION_SCANNED: " + String(sessionScanned) + "\n";  // Feature 6
    resp += "SESSION_DENIED: " + String(sessionDenied) + "\n";
    resp += "SESSION_OFFLINE: " + String(sessionOffline) + "\n";
    
    webServer.send(200, "text/plain", resp);
    return;
  }

  // 17. STATUS (JSON legacy response mapping)
  if (cmd == "STATUS") {
    int rssi = WiFi.RSSI();
    String resp = "{\"STATUS\":\"" + String(isOnline ? "ONLINE" : "OFFLINE") +
                  "\",\"MODE\":\"" + currentMode +
                  "\",\"COURSE\":\"" + currentCourse +
                  "\",\"QUEUE\":" + String(offlineQueueCount()) +
                  ",\"DEVICE\":\"" + DEVICE_NAME +
                  "\",\"RSSI\":" + String(rssi) +
                  ",\"UPTIME\":" + String(millis() / 1000) +
                  ",\"HEAP\":" + String(ESP.getFreeHeap()) +
                  ",\"FS_USED\":" + String(LittleFS.usedBytes()) +
                  ",\"FS_TOTAL\":" + String(LittleFS.totalBytes()) +
                  ",\"SESSION_SCANNED\":" + String(sessionScanned) +
                  ",\"SESSION_DENIED\":" + String(sessionDenied) +
                  ",\"SESSION_OFFLINE\":" + String(sessionOffline) +
                  ",\"BATTERY\":\"100%\"}";
    webServer.send(200, "application/json", resp);
    return;
  }

  // 18. GETCONFIG
  if (cmd == "GETCONFIG") {
    String resp = "ENABLE_FP=" + String(enableFingerprint ? "1" : "0") +
                  ",ENABLE_RFID=" + String(enableRfid ? "1" : "0") +
                  ",ENABLE_RC522=" + String(enableRfid ? "1" : "0") +
                  ",ENABLE_LCD=" + String(enableLcd ? "1" : "0") +
                  ",ENABLE_HW084=" + String(enableHw084 ? "1" : "0") +
                  ",ENABLE_FACE=" + String(enableFace ? "1" : "0") +
                  ",MFA=" + String(requireMfa ? "1" : "0") +
                  ",ENROLL_COUNT=" + String(enrollCount) +
                  ",SYSTEM_PIN=" + SYSTEM_PIN +
                  ",COURSE=" + currentCourse +
                  ",MODE=" + currentMode +
                  ",ONLINE=" + String(isOnline ? "1" : "0");
    webServer.send(200, "text/plain", resp);
    return;
  }

  // 19. SETCONFIG <cfg>
  if (cmd.startsWith("SETCONFIG ")) {
    String cfg = cmd.substring(10);
    cfg.trim();
    espPrintln("[OTA] Updating runtime parameters: " + cfg);
    
    preferences.begin("config", false);
    int startIdx = 0;
    while (startIdx < (int)cfg.length()) {
      int commaIdx = cfg.indexOf(',', startIdx);
      if (commaIdx < 0) commaIdx = cfg.length();
      String pair = cfg.substring(startIdx, commaIdx);
      int eqIdx = pair.indexOf('=');
      if (eqIdx > 0) {
        String key = pair.substring(0, eqIdx); key.trim();
        String val = pair.substring(eqIdx + 1); val.trim();
        preferences.putString(key.c_str(), val);
        
        // Runtime variables mapping
        if (key == "ENABLE_FP") { enableFingerprint = (val == "1"); }
        if (key == "ENABLE_RFID" || key == "ENABLE_RC522") { enableRfid = (val == "1"); }
        if (key == "ENABLE_LCD") { enableLcd = (val == "1"); }
        if (key == "ENABLE_HW084") { enableHw084 = (val == "1"); }
        // HW084_THRESHOLD retired — HW-084 now uses onboard ESP32 clock
        if (key == "ENABLE_FACE") { enableFace = (val == "1"); }
        if (key == "MFA") { requireMfa = (val == "1"); }
        if (key == "ENROLL_COUNT") { enrollCount = val.toInt(); }
        if (key == "SYSTEM_PIN") {
          SYSTEM_PIN = val;
        }
      }
      startIdx = commaIdx + 1;
    }
    preferences.end();
    webServer.send(200, "text/plain", "ACK: Parameters set successfully.");
    printMenu();
    return;
  }

  // 20. GETSECURITY
  if (cmd == "GETSECURITY") {
    webServer.send(200, "text/plain", "plaintext|");
    return;
  }

  // 21. SETCOURSES
  if (cmd.startsWith("SETCOURSES ")) {
    String list = cmd.substring(11);
    list.trim();
    preferences.begin("courses", false);
    preferences.putString("list", list);
    preferences.end();
    webServer.send(200, "text/plain", "ACK: Courses lists saved.");
    return;
  }

  // 22. SETSECURITY <params>
  if (cmd.startsWith("SETSECURITY ")) {
    String params = cmd.substring(12);
    params.trim();
    int p = params.indexOf('|');
    String method = (p >= 0) ? params.substring(0, p) : params;
    String token  = (p >= 0) ? params.substring(p + 1) : "";
    espPrintln("[OTA] Security update -> Method: " + method);
    preferences.begin("security", false);
    preferences.putString("method", method);
    preferences.putString("token", token);
    preferences.end();
    webServer.send(200, "text/plain", "ACK: Security set to " + method);
    return;
  }

  // 23. KEYPAD <key>
  if (cmd.startsWith("KEYPAD ")) {
    pendingKeypadKey = cmd.substring(7);
    pendingKeypadKey.trim();
    espPrintln("[KEYPAD] Virtual key pressed: " + pendingKeypadKey);
    webServer.send(200, "text/plain", "ACK: KEYPAD " + pendingKeypadKey);
    return;
  }

  // 24. SCANWIFI
  if (cmd == "SCANWIFI") {
    espPrintln("[OTA] Scanning Wi-Fi...");
    int n = WiFi.scanNetworks();
    String resp = "[";
    for (int i = 0; i < n; ++i) {
      if (i > 0) resp += ",";
      resp += "\"" + WiFi.SSID(i) + "\"";
    }
    resp += "]";
    webServer.send(200, "application/json", resp);
    return;
  }

  // 25. SCANBT
  if (cmd == "SCANBT") {
    webServer.send(200, "application/json", "[]");
    espPrintln("[OTA] BLE not compiled in. Enable USE_BLE in firmware.");
    return;
  }

  // 26. DOWNLOAD_FP_TEMPLATES
  if (cmd == "DOWNLOAD_FP_TEMPLATES") {
#ifdef USE_FINGERPRINT
    uint16_t count = 0;
    if (fpOK) {
      finger.getTemplateCount();
      count = finger.templateCount;
    }
    espPrintln("[OTA] Template count requested: " + String(count));
    webServer.send(200, "text/plain", "ACK: " + String(count) + " templates stored on sensor.");
#else
    webServer.send(200, "text/plain", "ERR: Fingerprint not compiled in.");
#endif
    return;
  }

  // 27. UPLOAD_FP_TEMPLATES
  if (cmd == "UPLOAD_FP_TEMPLATES") {
#ifdef USE_FINGERPRINT
    uint16_t count = 0;
    if (fpOK) {
      finger.getTemplateCount();
      count = finger.templateCount;
    }
    espPrintln("[OTA] Ready to receive templates. Current: " + String(count));
    webServer.send(200, "text/plain", "ACK: ESP ready. " + String(count) + " templates currently stored.");
#else
    webServer.send(200, "text/plain", "ERR: Fingerprint not compiled in.");
#endif
    return;
  }

  // 28. LISTFS
  if (cmd == "LISTFS") {
    String resp = "LittleFS Files:\n";
    File root = LittleFS.open("/");
    File file = root.openNextFile();
    int count = 0;
    while (file) {
      resp += "  " + String(file.name()) + " (" + String(file.size()) + " bytes)\n";
      file = root.openNextFile();
      count++;
    }
    if (count == 0) resp += "  (empty)\n";
    resp += "Total: " + String(count) + " file(s)";
    webServer.send(200, "text/plain", resp);
    return;
  }

  // Unknown Web API OTA Injector Command
  webServer.send(200, "text/plain", "ACK: Request ignored. Unknown Command: " + cmd);
}

// ============================================================
// SYSTEM RUNTIME INITIALIZATION
// ============================================================
bool initSDCard() {
  espPrintln("[SD] Initializing MicroSD card on shared SPI bus...");
  pinMode(SD_CS_PIN, OUTPUT);
  digitalWrite(SD_CS_PIN, HIGH); // deselect initially
  
  // Re-establish SPI bus with 1 MHz clock for stable breadboard signaling
  SPI.begin(18, 19, 23, -1);
  SPI.setFrequency(1000000); 
  delay(10);
  
  if (!SD.begin(SD_CS_PIN, SPI, 1000000)) {
    espPrintln("[SD] MicroSD Card initialization FAILED!");
    sdCardReady = false;
    #ifdef USE_FINGERPRINT
    sdFpReady = false;
    #endif
    return false;
  }
  
  sdCardReady = true;
  #ifdef USE_FINGERPRINT
  sdFpReady = true;
  #endif
  espPrintln("[SD] MicroSD Card ready. Total space: " + String((uint64_t)(SD.totalBytes() / (1024 * 1024))) + " MB");
  return true;
}

void setup() {
  Serial.begin(115200);
  delay(1000);

  // Disable brownout detector — fingerprint sensor + WiFi TX peak current
  // can trigger a false brownout reset (POWERON_RESET) on marginal supplies.
  WRITE_PERI_REG(RTC_CNTL_BROWN_OUT_REG, 0);

  bootMillis = millis();

  espPrintln("\n\n===========================================");
  espPrintln("  SENTINEL SWARM — BIOMETRICS ONLY BOOTING  ");
  espPrintln("===========================================");
  espPrintln("[POWER] If sensors fail when all components are connected:");
  espPrintln("[POWER] ESP32 onboard 3.3V regulator may be overloaded.");
  espPrintln("[POWER] Use external 3.3V supply for peripherals.");

  pinMode(BLUE_LED, OUTPUT);
  pinMode(FINGER_VOUT, INPUT);
  pinMode(BLUE_LED, OUTPUT);
  digitalWrite(BLUE_LED, LOW);

#ifdef USE_FINGERPRINT
  // FM10A VCC via MOSFET on D5 — power on here (HIGH = MOSFET on = VCC)
  pinMode(FP_MOSFET, OUTPUT);
  digitalWrite(FP_MOSFET, HIGH);
  delay(50);
#endif

  // ═══════════════════════════════════════════════════════════════
  // FM10A DETECTION FIRST — before I2C/SPI/Keypad init, so the
  // 3.3V rail is cleanest. Also hold MFRC522 in reset (D22 LOW)
  // to cut its ~30mA draw + 13.56MHz RF interference during init.
  // ═══════════════════════════════════════════════════════════════
  #ifdef USE_FINGERPRINT
  espPrintln("[SENSOR] Power stabilization delay (2s)...");
  delay(2000);
  // Power-cycle FM10A via MOSFET for clean startup
  digitalWrite(FP_MOSFET, LOW);
  delay(300);
  digitalWrite(FP_MOSFET, HIGH);
  delay(500);
  espPrintln("[SENSOR] Initializing FM10A biometrics engine...");
  fpOK = false;

  espPrintln("[SENSOR] FM10A TouchOut (D35) = " + String(digitalRead(FINGER_VOUT)) + " (0=powered likely)");

  // Try saved baud rate first, then scan
  int bauds[] = {fpBaudRate, 57600, 9600, 19200, 115200, 4800, 38400};
  for (int i = 0; i < 7 && !fpOK; i++) {
    if (i > 0 && bauds[i] == fpBaudRate) continue; // skip duplicate
    mySerial.begin(bauds[i], SERIAL_8N1, FINGER_RX2, FINGER_TX2);
    mySerial.setTimeout(2000);
    finger.begin(bauds[i]);
    delay(300);
    while (mySerial.available()) mySerial.read();
    // Send 0xEF sync bytes to reset FM10A's UART state machine
    for (int j = 0; j < 15; j++) {
      mySerial.write(0xEF);
      delay(1);
    }
    mySerial.flush();
    delay(100);
    while (mySerial.available()) mySerial.read();
    if (finger.verifyPassword()) {
      fpOK = true;
      fpBaudRate = bauds[i];
      // Save working baud rate to flash
      preferences.begin("fingerprint", false);
      preferences.putInt("baud", fpBaudRate);
      preferences.end();
      espPrintln("[SENSOR] FM10A found at " + String(fpBaudRate) + " baud (saved).");
      break;
    }
    mySerial.end();
    espPrintln("[SENSOR] No response at " + String(bauds[i]) + " baud.");
    delay(50);
  }

  if (!fpOK) {
    espPrintln("[SENSOR] ⚠ FM10A NOT DETECTED");
    espPrintln("[SENSOR] If it works alone but not with all components connected:");
    espPrintln("[SENSOR] → 3.3V rail overloaded. Use external 3.3V supply.");
    espPrintln("[SENSOR] → Verify wiring: VCC=3.3V, GND=GND, RX→D17, TX→D16, T-out→D35");
    espPrintln("[SENSOR] Will retry after network init and periodically in loop().");
  }
  #endif

  #ifdef USE_KEYPAD
  initKeypad();
  #endif
  
  // Power SD module lines (SPI active CS standard)
  pinMode(SD_CS_PIN, OUTPUT);
  digitalWrite(SD_CS_PIN, HIGH);

  flashLED(1, 200);

  // 1. Mount LittleFS partition
  espPrintln("[SYSTEM] Mounting internal flash partition LittleFS...");
  if (!LittleFS.begin(true)) {
    espPrintln("[SYSTEM] LittleFS ERROR: Mount failed! Offline queue unavailable.");
  } else {
    int q = offlineQueueCount();
    espPrintln("[SYSTEM] LittleFS OK. Queue loaded: " + String(q) + " scan(s) cached.");
  }

  // Init SPI bus (shared by RFID + SD card) with stable 1 MHz clock for breadboards
  SPI.begin(18, 19, 23, -1);
  SPI.setFrequency(1000000);

  // Initialize general MicroSD card interface
  initSDCard();

#ifdef USE_RFID
  espPrintln("[SENSOR] Booting MFRC522 SPI RFID reader...");
  // Full hardware reset sequence (matching working Test_frid)
  pinMode(RST_PIN, OUTPUT);
  digitalWrite(RST_PIN, LOW);
  delay(100);
  digitalWrite(RST_PIN, HIGH);
  delay(100);
  mfrc522.PCD_Init();
  mfrc522.PCD_AntennaOn();
  byte rfidVer = mfrc522.PCD_ReadRegister(MFRC522::VersionReg);
  if (rfidVer == 0x00 || rfidVer == 0xFF) {
    espPrintln("[SENSOR] ⚠ MFRC522 NOT DETECTED on SPI bus!");
  } else {
    char vbuf[48]; snprintf(vbuf, sizeof(vbuf), "[SENSOR] MFRC522 detected (firmware v0x%02X)", rfidVer);
    espPrintln(vbuf);
  }
  espPrintln("[SENSOR] To enable: SETCONFIG ENABLE_RFID=1");
#endif

#ifdef USE_FINGERPRINT
  // Init SD card for fingerprint template storage
  initFingerprintSD();
#endif
  logPowerSD("BOOT");

#ifdef USE_HW084
  espPrintln("[SENSOR] HW-084 RTC chip detected. Initializing DS3231...");
  Wire.begin(21, 22);
  if (rtc.begin()) {
    rtcOK = true;
    espPrintln("[SENSOR] DS3231 RTC found! Using as fallback clock when offline.");
  } else {
    espPrintln("[SENSOR] DS3231 RTC not found — will use onboard ESP32 clock.");
  }
#endif

  // Read SD config.txt if present (overrides NVRAM defaults)
  readSDConfig();

  // 3. Load dynamic Preferences parameters
  loadDeviceConfig();

  // 4. Initial network convergence + immediate time sync
  tryConnectWifi();
  if (isOnline) sendHeartbeat(); // sync clock before any scan

#ifdef USE_FINGERPRINT
  // 4.5b. Third FM10A detection pass — power rail fully stabilized after Wi-Fi
  if (!fpOK) {
    espPrintln("\n[SENSOR] Retrying FM10A after Wi-Fi (rail stabilized)...");
    int bauds3[] = {fpBaudRate, 57600, 9600, 19200, 115200, 4800};
    for (int i = 0; i < 6 && !fpOK; i++) {
      if (i > 0 && bauds3[i] == fpBaudRate) continue;
      mySerial.begin(bauds3[i], SERIAL_8N1, FINGER_RX2, FINGER_TX2);
      finger.begin(bauds3[i]);
      delay(800);
      while (mySerial.available()) mySerial.read();
      if (finger.verifyPassword()) {
        fpOK = true;
        espPrintln("[SENSOR] FM10A found at " + String(bauds3[i]) + " baud (third pass).");
        break;
      }
      mySerial.end();
    }
    if (!fpOK) {
      espPrintln("[SENSOR] ⚠ FM10A still not detected after rail stabilization.");
      espPrintln("[SENSOR] → External 3.3V regulator likely required for reliable operation.");
    }
  }
#endif

#ifdef USE_LCD
  espPrintln("[LCD] Booting virtual/physical LCD display...");
  Wire.begin(21, 22);
  byte addrs[] = {0x27, 0x3F, 0x3E, 0x23, 0x21, 0x25, 0x26, 0x20};
  byte lcdAddr = 0;
  for (int i = 0; i < 8; i++) {
    Wire.beginTransmission(addrs[i]);
    if (Wire.endTransmission() == 0) { lcdAddr = addrs[i]; break; }
  }
  if (lcdAddr == 0) {
    espPrintln("[LCD] ⚠ No I2C LCD found. Checked: 0x27,0x3F,0x3E,0x23,0x21,0x25,0x26,0x20");
    espPrintln("[LCD] Verify SDA/D21 and SCL/D22 wiring, or run I2C scanner to find the address.");
  } else {
    char buf[40]; snprintf(buf, sizeof(buf), "[LCD] Found display at 0x%02X", lcdAddr);
    espPrintln(buf);
    lcdPtr = new LiquidCrystal_I2C(lcdAddr, LCD_COLS, LCD_ROWS);
    lcdPtr->init();
    lcdPtr->backlight();
    lcdOK = true;
    espPrintln("[LCD] Display initialized (" + String(LCD_COLS) + "x" + String(LCD_ROWS) + ") successfully.");
  }
#endif

  // 5. Build local http server
  webServer.on("/cmd", HTTP_GET,     handleDashboardCommand);
  webServer.on("/cmd", HTTP_OPTIONS, []() {
    webServer.sendHeader("Access-Control-Allow-Origin", "*");
    webServer.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
    webServer.sendHeader("Access-Control-Allow-Headers", "*");
    webServer.send(204);
  });
  webServer.on("/logs", HTTP_GET,    handleGetLogs);
  webServer.on("/logs", HTTP_OPTIONS, []() {
    webServer.sendHeader("Access-Control-Allow-Origin", "*");
    webServer.sendHeader("Access-Control-Allow-Methods", "GET, OPTIONS");
    webServer.sendHeader("Access-Control-Allow-Headers", "*");
    webServer.send(204);
  });
  webServer.on("/sdfile", HTTP_GET,  handleSDFile);
  webServer.begin();
  espPrintln("[WEB] Local HTTP Web Server dashboard route listening on Port 80.");

  printMenu();

  // Show PIN lock screen on boot
  if (lcdOK && systemLocked) {
    lcdClear();
    lcdWrite(0, "== LOCKED ==");
    lcdWrite(1, "Enter PIN:");
    espPrintln("\n========================================");
    espPrintln("  SYSTEM LOCKED - Enter PIN to unlock   ");
    espPrintln("  Enter 123 then press # or D");
    espPrintln("========================================");
  }
}

// ============================================================
// MAIN SYSTEM LOOP
// ============================================================
void loop() {
  // 0. Physical 4×4 keypad scan — first, before HTTP/web processing
  #ifdef USE_KEYPAD
  {
    char k = scanKeypad();
    if (k && k != lastKey) {
      if (millis() < 3000) {
        lastKey = k;
        lastKeyTime = millis();
        if (pendingKeypadKey.length() == 0) espPrintln("[KEYPAD] Boot guard - ignoring '" + String(k) + "'");
      } else {
        lastKey = k;
        lastKeyTime = millis();
        pendingKeypadKey = String(k);
      espPrintln("[KEYPAD] Key pressed: " + String(k));
      // Temporary diagnostic: print the intended key vs what's received
      if (k == '2') espPrintln("[DIAG] Got '2' - if you pressed '1', mapping needs fixing");
      if (k == '5') espPrintln("[DIAG] Got '5' - if you pressed '2', mapping needs fixing");
      if (k == '8') espPrintln("[DIAG] Got '8' - if you pressed '3', mapping needs fixing");
      if (k == 'B') espPrintln("[DIAG] Got 'B' - if you pressed '4', mapping needs fixing");
      }
    }
    if (!k && millis() - lastKeyTime > 200) lastKey = 0;
  }
  #endif

  // ── PIN entry: check FIRST (before display, web, server, etc.) ──────
  if (systemLocked) {
    String pinInput = "";
    if (pendingKeypadKey.length() > 0) {
      pinInput = pendingKeypadKey;
      pendingKeypadKey = "";
    }
    if (Serial.available() > 0) {
      String ser = Serial.readStringUntil('\n');
      ser.trim();
      if (ser.length() > 0) pinInput = ser;
    }
    if (pinInput.length() > 0) {
      espPrintln("[PIN] Input: " + pinInput);
      if (pinInput.length() > 1) {
        if (pinInput == SYSTEM_PIN || pinInput == SYSTEM_PIN + "#" || pinInput == SYSTEM_PIN + "D") {
          systemLocked = false; pinBuffer = "";
          lcdClear(); lcdWrite(0, "== UNLOCKED =="); lcdWrite(1, "System Ready");
          espPrintln("\n[PIN] Correct! System unlocked.");
          delay(1000); lcdClear(); lcdWrite(0, "ATTENDANCE SCAN");
          lcdWrite(1, currentCourse.substring(0, LCD_COLS));
          lcdWrite(2, isOnline ? "ONLINE" : "OFFLINE");
          lcdWrite(3, "Keys: A=Course"); printMenu();
        } else {
          espPrintln("[PIN] Wrong PIN!"); lcdWrite(1, "Wrong PIN!"); pinBuffer = ""; delay(1200); lcdWrite(1, "Enter PIN:");
        }
      }
      else if (pinInput.length() == 1 && pinInput.charAt(0) >= '0' && pinInput.charAt(0) <= '9') {
        if (pinBuffer.length() < 10) { pinBuffer += pinInput; String stars = ""; for (int _i = 0; _i < pinBuffer.length(); _i++) stars += "*"; lcdWrite(1, "PIN:" + stars); espPrintln("[PIN] Buffer: " + pinBuffer); }
      }
      else if (pinInput == "*" || pinInput == "C" || pinInput == "c") {
        pinBuffer = ""; lcdWrite(1, "PIN cleared"); espPrintln("[PIN] Cleared");
      }
      else if (pinInput == "#" || pinInput == "D") {
        if (pinBuffer == SYSTEM_PIN) {
          systemLocked = false; pinBuffer = "";
          lcdClear(); lcdWrite(0, "== UNLOCKED =="); lcdWrite(1, "System Ready");
          espPrintln("\n[PIN] Correct! System unlocked.");
          delay(1000); lcdClear(); lcdWrite(0, "ATTENDANCE SCAN");
          lcdWrite(1, currentCourse.substring(0, LCD_COLS));
          lcdWrite(2, isOnline ? "ONLINE" : "OFFLINE");
          lcdWrite(3, "Keys: A=Course"); printMenu();
        } else { espPrintln("[PIN] Wrong PIN!"); lcdWrite(1, "Wrong PIN!"); pinBuffer = ""; delay(1200); lcdWrite(1, "Enter PIN:"); }
      }
    }
    delay(20);
    return;
  }

  // ── Display sleep after 60s inactivity ──────────────────────────────
  if (displaySleeping) {
    if (pendingKeypadKey.length() > 0) {
      displaySleeping = false;
      lastActivity = millis();
      #ifdef USE_LCD
      if (lcdOK) {
        restoreI2cForLcd();
        lcdPtr->backlight();
        reclaimRstForRfid();
        for (int r = 0; r < 4; r++) lcdWrite(r, lcdLine[r]);
      }
      #endif
      espPrintln("[DISPLAY] Woke from sleep (keypad press).");
      printMenu();
    }
  } else if (millis() - lastActivity >= displaySleepTimeout) {
    displaySleeping = true;
    #ifdef USE_LCD
    if (lcdOK) {
      restoreI2cForLcd();
      lcdPtr->clear();
      lcdPtr->noBacklight();
      reclaimRstForRfid();
    }
    #endif
    espPrintln("[DISPLAY] Sleeping — 60s idle. Press any key to wake.");
    logPowerSD("DISPLAY_SLEEP");
  }

  webServer.handleClient();

  // 1. Wi-Fi watchdog Auto-Reconnector
  if (WiFi.status() != WL_CONNECTED && millis() - lastWifiRetry >= WIFI_RETRY_MS) {
    espPrintln("\n[WIFI] Watchdog: Lost server ping. Attempting reconnection...");
    tryConnectWifi();
    lastWifiRetry = millis();
  }

  // 1.5. Server reachability watchdog — retry when WiFi is up but server is unreachable
  if (WiFi.status() == WL_CONNECTED && !isOnline && millis() - lastServerRetry >= SERVER_RETRY_MS) {
    espPrintln("[SERVER] Watchdog: Attempting server reachability check...");
    WiFiClient testClient;
    HTTPClient testHttp;
    testHttp.setTimeout(3000);
    testHttp.begin(testClient, heartbeatURL + "?ping=1&hardware_key=" + hardwareKey);
    int code = testHttp.GET();
    if (code > 0) {
      isOnline = true;
      espPrintln("[SERVER] Reachable again! Switching to ONLINE mode.");
      syncOfflineQueue();
      flashLED(2, 100);
    } else {
      espPrintln("[SERVER] Still unreachable (HTTP " + String(code) + "). Retrying in " + String(SERVER_RETRY_MS/1000) + "s.");
    }
    testHttp.end();
    lastServerRetry = millis();
  }

  // 2. Scheduled Heartbeat Ping (registers IP + receives server clock)
  if (millis() - lastHeartbeatTime >= HEARTBEAT_INTERVAL) {
    sendHeartbeat();
    lastHeartbeatTime = millis();
  }

#ifdef USE_FINGERPRINT
  // 2.5. Periodic FM10A re-detection (15s intervals, backs off after 2 min)
  static unsigned long lastFpRetry = 0;
  static int fpRetryCount = 0;
  unsigned long fpInterval = (fpRetryCount < 8) ? 15000UL : 60000UL;
  if (!fpOK && millis() - lastFpRetry > fpInterval) {
    lastFpRetry = millis();
    fpRetryCount++;
    espPrintln("[SENSOR] Periodic FM10A re-detection (retry #" + String(fpRetryCount) + ")...");
    // Power-cycle FM10A via MOSFET before retry
    digitalWrite(FP_MOSFET, LOW);
    delay(300);
    digitalWrite(FP_MOSFET, HIGH);
    delay(500);
    // Try saved baud rate first, then fallback list
    int baudsP[] = {fpBaudRate, 57600, 9600, 19200, 115200, 4800};
    for (int i = 0; i < 6 && !fpOK; i++) {
      if (i > 0 && baudsP[i] == fpBaudRate) continue;
      mySerial.begin(baudsP[i], SERIAL_8N1, FINGER_RX2, FINGER_TX2);
      finger.begin(baudsP[i]);
      for (int w = 0; w < 12; w++) { delay(50); webServer.handleClient(); }
      while (mySerial.available()) mySerial.read();
      if (finger.verifyPassword()) {
        fpOK = true;
        espPrintln("[SENSOR] FM10A found at " + String(baudsP[i]) + " baud (periodic retry).");
        break;
      }
      mySerial.end();
      webServer.handleClient();
    }
  }
#endif

  // 3. Scheduled Offline Queue Sync
  if (isOnline && WiFi.status() == WL_CONNECTED && millis() - lastSyncAttempt >= SYNC_INTERVAL) {
    syncOfflineQueue();
    lastSyncAttempt = millis();
  }

  // Feature 3: Midnight auto-reset — clear course and session stats at new day
  static int lastDay = -1;
  {
    struct tm timeinfo;
    time_t now;
    time(&now);
    localtime_r(&now, &timeinfo);
    if (timeSynced && lastDay != -1 && timeinfo.tm_mday != lastDay) {
      espPrintln("[AUTO] New day detected. Resetting session stats.");
      sessionScanned = 0;
      sessionDenied  = 0;
      sessionOffline = 0;
      currentCourse  = "Unassigned Session";
      currentMode    = "ATTENDANCE_MODE";
      printMenu();
    }
    if (timeSynced) lastDay = timeinfo.tm_mday;
  }

  // 4. Process OTA Web-Triggered Enrollment requests
  if (currentMode == "ENROLL_MODE" && pendingEnrollSlot != -1) {
    runEnrollment(pendingEnrollSlot);
    pendingEnrollSlot = -1;
    currentMode = "ATTENDANCE_MODE";
    printMenu();
  }

  // 4.2. Process Web-triggered RFID scan (set by BEEP command)
  #ifdef USE_RFID
  if (pendingRfidScan) {
    runRfidScanForWeb();
    // pendingRfidScan cleared inside runRfidScanForWeb
    printMenu();
  }
  #endif

  // 4.6. Process remaining keypad keys (unlocked)
  if (pendingKeypadKey.length() > 0) {
    lastActivity = millis();
    String key = pendingKeypadKey;
    pendingKeypadKey = "";

    // ── Route key to course-picker if picker is open ──────────────────
    if (courseMenuOpen) {
      coursePickerKey(key);
    }
    // ── Await mode choice after course selected ───────────────────────
    else if (awaitModeChoice) {
      if (key == "A") {
        awaitModeChoice = false;
        currentMode = "ATTENDANCE_MODE";
        espPrintln("[MODE] Attendance mode started.");
        lcdClear();
        lcdWrite(0, "ATTENDANCE SCAN");
        lcdWrite(1, currentCourse.substring(0, LCD_COLS));
        lcdWrite(2, isOnline ? "ONLINE" : "OFFLINE");
        lcdWrite(3, "A=Course");
        printMenu();
      } else if (key == "B") {
        awaitModeChoice = false;
        currentMode = "ENROLL_MODE";
        espPrintln("[MODE] Enrollment mode started.");
        runEnrollment(-1);
        currentMode = "ATTENDANCE_MODE";
        printMenu();
      } else if (key == "*") {
        awaitModeChoice = false;
        espPrintln("[PICKER] Mode selection cancelled.");
        lcdClear();
        lcdWrite(0, "Cancelled.");
        delay(600);
        printMenu();
      }
    }
    // ── A → Open course picker ────────────────────────────────────────
    else if (key == "A") {
      openCoursePicker();
    }
    // ── Digit keys → T9 multi-tap / RFID enroll trigger ──────────
    else if (key.length() == 1 && key.charAt(0) >= '0' && key.charAt(0) <= '9') {
      // Key '0' with empty T9 buffer → enter RFID enrollment
      if (key == "0" && t9Buffer.length() == 0 && !courseMenuOpen) {
#ifdef USE_RFID
        runRfidEnrollment();
#endif
      } else {
        t9HandleKey(key.charAt(0));
      }
    }
    // ── * → Backspace T9 or go back ──────────────────────────────────
    else if (key == "*") {
      if (t9Buffer.length() > 0) {
        t9Buffer = t9Buffer.substring(0, t9Buffer.length()-1);
        String preview = t9Buffer.length() > 0 ? t9Buffer : "_";
        lcdWrite(0, "T9 Input:");
        lcdWrite(1, preview);
        espPrintln("[T9] Del. Buffer: '" + t9Buffer + "'");
      } else {
        lcdClear();
        lcdWrite(0, "[*] Back");
      }
    }
    // ── D → Confirm T9 buffer as course if not empty ──────────────────
    else if (key == "D") {
      if (t9Buffer.length() > 0) {
        currentCourse = t9Buffer;
        t9Buffer = "";
        t9LastKey = 0;
        currentMode = "ATTENDANCE_MODE";
        espPrintln("[COURSE] T9 confirmed: " + currentCourse);
        lcdClear();
        lcdWrite(0, "ATTENDANCE SCAN");
        lcdWrite(1, currentCourse.substring(0, LCD_COLS));
        lcdWrite(2, isOnline ? "ONLINE" : "OFFLINE");
        lcdWrite(3, "Ready to scan...");
        flashLED(2, 100);
        printMenu();
      } else {
        lcdClear();
        lcdWrite(0, "[D] Confirm");
        lcdWrite(1, currentCourse);
      }
    }
    // ── B → Show session stats on LCD (Feature 5) ────────────────────
    else if (key == "B") {
      lcdClear();
      lcdWrite(0, "SESSION STATS");
      lcdWrite(1, "Scanned: " + String(sessionScanned));
      lcdWrite(2, "Denied:  " + String(sessionDenied));
      lcdWrite(3, "Offline: " + String(sessionOffline));
      espPrintln("[KEYPAD] B - Session stats: " + String(sessionScanned) + " scanned, " + String(sessionDenied) + " denied, " + String(sessionOffline) + " offline");
    } else if (key == "C") {
      lcdClear();
      lcdWrite(0, isOnline ? "ONLINE" : "OFFLINE");
      lcdWrite(1, "IP:" + WiFi.localIP().toString());
      lcdWrite(2, "RSSI:" + String(WiFi.RSSI()) + "dBm");
      lcdWrite(3, "Heap:" + String(ESP.getFreeHeap()/1024) + "KB free");
    }
    // ── # → force sync ───────────────────────────────────────────────
    else if (key == "#") {
      espPrintln("[KEYPAD] # pressed — forcing offline sync.");
      syncOfflineQueue();
      lcdClear();
      lcdWrite(0, "[#] Force Sync");
      lcdWrite(1, isOnline ? "Syncing..." : "OFFLINE!");
    } else {
      lcdClear();
      lcdWrite(0, "Key: " + key);
      lcdWrite(1, currentCourse.substring(0, LCD_COLS));
    }
  }

  // 5. Interactive Serial Commands Console
  if (Serial.available() > 0) {
    char cmd = Serial.read();
    
    // Clean NL / CR spaces
    if (cmd == '\n' || cmd == '\r' || cmd == ' ' || cmd == '\t' || cmd == '.') {
      // Discard silently
    } 
    else if (cmd == 'E' || cmd == 'e') {
      currentMode = "ENROLL_MODE";
      runEnrollment(-1);
      currentMode = "ATTENDANCE_MODE";
      printMenu();
    } 
    else if (cmd == 'D' || cmd == 'd') {
      deleteFingerprintSlot();
      printMenu();
    } 
    else if (cmd == 'C' || cmd == 'c') {
      manualCourseChange();
      printMenu();
    } 
    else if (cmd == 'S' || cmd == 's') {
      printDiagnostics();
      printMenu();
    } 
    else if (cmd == 'X' || cmd == 'x') {
      wipeEntireDatabase();
      printMenu();
    } 
    else if (cmd == 'Y' || cmd == 'y') {
      espPrintln("\n[SYNC] Initiated manual background queue push...");
      syncOfflineQueue();
      printMenu();
    } 
    else if (cmd == 'O' || cmd == 'o') {
      dumpOfflineQueue();
      printMenu();
    } 
    else if (cmd == 'K' || cmd == 'k') {
      clearOfflineQueue();
      printMenu();
    } 
    else if (cmd == 'R' || cmd == 'r') {
      espPrintln("\n[REBOOT] Manual software reboot initiated...");
      delay(500);
      ESP.restart();
    } 
    else if (cmd == 'A' || cmd == 'a') {
      openCoursePicker();  // uses serial fallback when offline
    }
    else if (cmd == 'I' || cmd == 'i') {
#ifdef USE_RFID
      runRfidEnrollment();
      printMenu();
#else
      espPrintln("[ERROR] RFID not enabled (USE_RFID not defined).");
#endif
    }
    else if (cmd == 'F' || cmd == 'f') {
      backupAllFpTemplatesToServer();
    }
    else if (cmd == 'G' || cmd == 'g') {
      restoreFpTemplatesFromServer();
    }
    // Feature 4: Serial 'T' — show session totals
    else if (cmd == 'T' || cmd == 't') {
      espPrintln("\n=== SESSION TOTALS ===");
      espPrintln("  Course:  " + currentCourse);
      espPrintln("  Mode:    " + currentMode);
      espPrintln("  Scanned: " + String(sessionScanned));
      espPrintln("  Denied:  " + String(sessionDenied));
      espPrintln("  Offline: " + String(sessionOffline));
      espPrintln("  Queue:   " + String(offlineQueueCount()));
      espPrintln("  Time:    " + getTimestamp());
      espPrintln("======================");
    }
  }

  // Auto-commit T9 if key has been idle for T9_TIMEOUT (no action needed
  // — the buffer just stays until D or next key clears it)
  t9Tick();

  // 6. Background scan polling (skip when locked or enrolling RFID)
  if (!systemLocked && !rfidEnrollMode && currentMode == "ATTENDANCE_MODE") {
    #if defined(USE_FINGERPRINT) && defined(USE_RFID)
    if (!fpOK && !enableRfid) {
      static unsigned long lastSensorWarn = 0;
      if (millis() - lastSensorWarn > 10000) {
        lastSensorWarn = millis();
        espPrintln("[ATTEND] ⚠ Both sensors disabled. FP=" + String(fpOK?"OK":"FAIL") + " RFID=" + String(enableRfid?"ON":"OFF"));
        espPrintln("[ATTEND] Enable RFID: SETCONFIG ENABLE_RFID=1  |  Fix FM10A: external 3.3V supply");
      }
    }
    #endif
    scanForAttendance();
  }

  // 7. Clock display — update only line 0 with system time (doesn't overwrite other lines)
  static unsigned long lastClockShow = 0;
  if (millis() - lastClockShow >= 5000) {
    lastClockShow = millis();
    String sysTime = getTimestamp();
    if (sysTime.length() >= 19) sysTime = sysTime.substring(11, 19);
    else sysTime = "No sync";
    lcdWrite(0, sysTime);
  }

  delay(20);
}
